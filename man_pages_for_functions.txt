------------------------------------------------------------------------------
NAME
       cvector_create - create a vector with default memory management

SYNOPSIS
       #include <cvector.h>

       cvec cvector_create(size_t elem_size, char **err);

DESCRIPTION
       Creates a new vector container that stores elements of the specified size.
       The vector starts with a minimum capacity of 4 elements and automatically
       resizes as needed. Uses standard malloc/free for memory management.

       This is a static inline wrapper around cvector_create_full() with NULL
       memory management procedures.

PARAMETERS
       elem_size  Size of each element in bytes (must be > 0)
       err        Optional pointer to receive error string on failure
                  (pass NULL to ignore)

RETURN VALUE
       Returns pointer to newly created vector on success
       Returns NULL on failure (error string set in *err if provided)

NOTES
       Initial capacity is 4 elements
       Capacity doubles when full, halves when < 1/4 filled
       Must be destroyed with cvector_destroy() when done

SEE ALSO
       cvector_create_full, cvector_destroy, cvec_construct

NAME
       cvector_create_full - create a vector with custom memory management

SYNOPSIS
       #include <cvector.h>

       cvec cvector_create_full(size_t elem_size, 
                                ccol_memmgmt_procs_t *mmgmt_procs,
                                char **err);

DESCRIPTION
       Creates a new vector container with full customization of memory 
       management. The vector stores elements of the specified size and 
       automatically manages capacity.

PARAMETERS
       elem_size     Size of each element in bytes (must be > 0)
       mmgmt_procs   Custom memory management procedures, or NULL for 
                     default malloc/free
       err           Optional pointer to receive error string on failure

RETURN VALUE
       Returns pointer to newly created vector on success
       Returns NULL on failure

NOTES
       Initial capacity is 4 elements
       Capacity scaling: 2x when full, 0.5x when < 1/4 filled
       Minimum capacity: 4 elements

SEE ALSO
       cvector_create, cvector_destroy

------------------------------------------------------------------------------
NAME
       cvector_push_back - append an element to the end of the vector

SYNOPSIS
       #include <cvector.h>

       ccol_retval_t cvector_push_back(cvec v, const void *new_elem);

DESCRIPTION
       Adds a new element to the end of the vector. If the vector is at capacity,
       it automatically grows by doubling its capacity. The element is copied 
       into the vector using an optimized assignment based on element size.

PARAMETERS
       v         Vector to append to
       new_elem  Pointer to element to append (must not be NULL)

RETURN VALUE
       ccol_success          - Element successfully appended
       ccol_invalid_args     - new_elem is NULL
       ccol_container_full   - Vector has reached max_elem_count
       ccol_not_enough_memory - Capacity expansion failed

NOTES
       Amortized O(1) complexity
       Element is copied into the vector
       Capacity doubles when full (2x scaling factor)
       Asserts if v is NULL

SEE ALSO
       cvector_pop_back, cvec_push, cvec_push_rvalue

------------------------------------------------------------------------------
NAME
       cvector_pop_back - remove and return the last element from the vector

SYNOPSIS
       #include <cvector.h>

       ccol_retval_t cvector_pop_back(cvec v, void *target_elem);

DESCRIPTION
       Removes the last element from the vector and copies it to the target 
       buffer. If the vector becomes less than 1/4 full, it automatically 
       shrinks by halving its capacity (minimum capacity is 4).

PARAMETERS
       v            Vector to pop from
       target_elem  Pointer to buffer to receive the element (must not be NULL)

RETURN VALUE
       ccol_success        - Element successfully popped
       ccol_invalid_args   - target_elem is NULL
       ccol_container_empty - Vector is empty

NOTES
       O(1) complexity
       Element is copied to target_elem
       Capacity halves when < 1/4 full (down to minimum of 4)
       Asserts if v is NULL

SEE ALSO
       cvector_push_back, cvec_pop

NAME
       cvector_reset - clear all elements and reset capacity

SYNOPSIS
       #include <cvector.h>

       void cvector_reset(cvec v);

DESCRIPTION
       Removes all elements from the vector and attempts to shrink the capacity
       back to the minimum (4 elements). If reallocation fails, the capacity
       remains unchanged but the element count is still reset to 0.

PARAMETERS
       v    Vector to reset

NOTES
       Element count becomes 0
       Capacity reset to minimum (4) if reallocation succeeds
       Does not free individual elements - caller must do this first if needed
       Asserts if v is NULL

SEE ALSO
       cvec_reset, cvector_destroy

------------------------------------------------------------------------------
NAME
       cvector_reserve - reserve capacity for future elements

SYNOPSIS
       #include <cvector.h>

       bool cvector_reserve(cvec v, size_t new_capacity_count);

DESCRIPTION
       Pre-allocates capacity to hold at least new_capacity_count elements,
       rounded up to the nearest power of two (minimum 4). Helps avoid multiple
       reallocations when the expected size is known in advance.

PARAMETERS
       v                    Vector to reserve capacity for
       new_capacity_count   Minimum number of elements to reserve

RETURN VALUE
       Returns true if reservation succeeds or capacity is already sufficient
       Returns false if allocation fails or new_capacity_count > max_elem_count

NOTES
       Does nothing if requested capacity <= current capacity
       Actual capacity may be larger due to power-of-2 rounding
       Does not change element count
       Asserts if v is NULL

SEE ALSO
       cvec_reserve

       cvector_append_array - append array of elements to vector

       Asserts if v is NULL

------------------------------------------------------------------------------
NAME
       cvector_find - find the index of the first occurrence of an element

SYNOPSIS
       #include <cvector.h>

       size_t cvector_find(cvec v, const void *elem,
                           ccol_comparison_proc_t cmp);

DESCRIPTION
       Performs a linear scan and returns the zero-based index of the first
       element that compares equal to *elem. When cmp is NULL the comparison
       is done with memcmp over the element size (byte-wise equality).

PARAMETERS
       v     Vector to search (must not be NULL)
       elem  Pointer to the value to search for
       cmp   Comparison function (int(*)(const void*,const void*)), or NULL
             to use memcmp

RETURN VALUE
       Zero-based index of the first matching element.
       ccol_invalid_size if no match is found or elem is NULL.

NOTES
       O(n) complexity
       Asserts if v is NULL
       Structs with padding bytes may not compare correctly when cmp is NULL

SEE ALSO
       cvec_find

NAME
       chmap_create - create a hash map with default settings

SYNOPSIS
       #include <chashmap.h>

       chmap chmap_create(size_t initial_bucket_array_size,
                          ccol_data_type key_type, 
                          ccol_data_type val_type, 
                          char **err);

DESCRIPTION
       Creates a hash map with default memory management and default hashing.
       The implementation strategy (open-addressing vs separate chaining) is
       automatically selected based on the key and value types.

       Open-addressing is used when both key and value are integral types <=8 
       bytes (excluding long double). Separate chaining is used for non-integral
       types or types >8 bytes.

PARAMETERS
       initial_bucket_array_size  Initial number of buckets (minimum 16)
       key_type                   Type of keys (determines implementation)
       val_type                   Type of values (determines implementation)
       err                        Optional pointer to receive error string

RETURN VALUE
       Returns pointer to newly created hash map on success
       Returns NULL on failure

NOTES
       Minimum bucket count is 16 (always power-of-2)
       Open-addressing: Fibonacci hashing for integers, XXHash64 for buffers
       Separate chaining: XXHash64 for all types
       Map must be destroyed with chmap_destroy() when done

SEE ALSO
       chmap_create_full, chmap_create_mp, chmap_create_ch, chmap_destroy

------------------------------------------------------------------------------
NAME
       chmap_create_full - create a hash map with full customization

SYNOPSIS
       #include <chashmap.h>

       chmap chmap_create_full(size_t initial_bucket_array_size,
                               ccol_data_type key_type, 
                               ccol_data_type val_type,
                               ccol_memmgmt_procs_t *mmgmt_procs,
                               ccol_hashing_proc_t custom_hashing_proc, 
                               char **err);

DESCRIPTION
       Creates a hash map with full control over initial size, types, memory
       management, and hashing function. The implementation strategy is
       automatically selected based on key and value types.

PARAMETERS
       initial_bucket_array_size  Initial number of buckets
       key_type                   Type of keys
       val_type                   Type of values
       mmgmt_procs                Custom memory management, or NULL
       custom_hashing_proc        Custom hash function, or NULL
       err                        Optional pointer to receive error string

RETURN VALUE
       Returns pointer to newly created hash map on success
       Returns NULL on failure

NOTES
       Implementation selection:
         - Open-addressing: Both key and value are integral types <=8 bytes
         - Separate chaining: Either key or value is non-integral or >8 bytes
       Integral types: char, short, int, long, long long (signed/unsigned),
                      float, double
       Excludes long double from open-addressing (can be 10-16 bytes)

SEE ALSO
       chmap_create, chmap_destroy

------------------------------------------------------------------------------
NAME
       chmap_insert_elem - insert a key-value pair into the hash map

SYNOPSIS
       #include <chashmap.h>

       ccol_retval_t chmap_insert_elem(chmap chm,
                                        const cmap_pair *key_pair,
                                        const cmap_pair *val_pair);

DESCRIPTION
       Inserts a key-value pair into the hash map. If the key already exists,
       the old value is replaced with the new value. Automatically resizes the
       map when load factor thresholds are exceeded.

PARAMETERS
       chm        Hash map to insert into
       key_pair   Key to insert (copied into map)
       val_pair   Value to associate with key (copied into map)

RETURN VALUE
       ccol_success               - New key inserted successfully
       ccol_key_already_present   - Key already existed; value updated successfully
       ccol_invalid_args          - chm, key_pair, or val_pair is NULL or size is 0
       ccol_container_full        - Map has reached max_elem_count
       ccol_not_enough_memory     - Memory allocation failed

NOTES
       Average O(1) complexity
       Key and value data are copied into the map
       If key exists, old value is freed and replaced
       Automatically resizes when load factor exceeded

SEE ALSO
       chmap_insert, chmap_delete_elem, chmap_get_elem_ref

------------------------------------------------------------------------------
NAME
       chmap_delete_elem - remove a key-value pair from the hash map

SYNOPSIS
       #include <chashmap.h>

       ccol_retval_t chmap_delete_elem(chmap chm, const cmap_pair *key_pair);

DESCRIPTION
       Removes the key-value pair with the specified key from the hash map.
       Frees memory for both key and value. Automatically shrinks the map when
       load factor falls below threshold.

PARAMETERS
       chm        Hash map to remove from
       key_pair   Key to remove

RETURN VALUE
       ccol_success        - Key removed successfully
       ccol_invalid_args   - chm or key_pair is NULL or key size is 0
       ccol_key_not_found  - Key does not exist in map

NOTES
       Average O(1) complexity
       Frees memory for key and value
       Open-addressing: shrinks when load factor < 0.25 (scale 0.5x)
       Separate chaining: shrinks when elem_count < (bucket_count + 1) / 8 (scale 0.25x)

SEE ALSO
       chmap_remove, chmap_insert_elem

------------------------------------------------------------------------------
NAME
       chmap_get_elem_ref - get reference to value by key

SYNOPSIS
       #include <chashmap.h>

       ccol_retval_t chmap_get_elem_ref(chmap chm,
                                         const cmap_pair *key_pair,
                                         cmap_pair **val_pair);

DESCRIPTION
       Retrieves a reference to the value associated with the given key. The
       returned reference points to internal storage and can be used to read
       or modify the value in-place.

PARAMETERS
       chm        Hash map to search

RETURN VALUE
       ccol_success        - Key found, *val_pair points to value
       ccol_invalid_args   - chm, key_pair, or val_pair is NULL or key size is 0
       ccol_key_not_found  - Key does not exist in map

NOTES
       Average O(1) complexity
       Returned reference is valid until next insert/delete/resize
       Reference can be used to modify value in-place

SEE ALSO
       chmap_get, chmap_get_ptr, chmap_insert_elem

SEE ALSO

------------------------------------------------------------------------------
NAME
       chmap_reset - clear all elements from the hash map

SYNOPSIS
       #include <chashmap.h>

       ccol_retval_t chmap_reset(chmap chm, size_t new_bucket_array_size);

DESCRIPTION
       Removes all key-value pairs from the hash map and destroys all internal
       data. Optionally resizes the bucket/slot array. The map remains usable
       after reset regardless of whether the resize succeeds.

PARAMETERS
       chm                      Hash map to reset
       new_bucket_array_size    New bucket count (0 = keep current size;
                                values < 16 are raised to 16; otherwise
                                rounded to the nearest power of two)

RETURN VALUE
       ccol_success           - All elements cleared (and resize succeeded if requested)
       ccol_not_enough_memory - Resize failed; elements are still cleared

NOTES
       O(n) complexity where n is number of elements
       All elements are destroyed regardless of return value
       Asserts if chm is NULL

SEE ALSO
       chmap_destroy, chmap_delete_elem

------------------------------------------------------------------------------
NAME
       cbmap_create - create a BST map with default settings

SYNOPSIS
       #include <cbstmap.h>

       cbmap cbmap_create(bool keys_are_signed_ints, char **err);

DESCRIPTION
       Creates a self-balancing binary search tree map (AVL tree) with default
       memory management and default comparison based on key signedness.

PARAMETERS
       keys_are_signed_ints  If true, treat integer keys as signed for comparison
       err                   Optional pointer to receive error string on failure

RETURN VALUE
       Returns pointer to newly created BST map on success
       Returns NULL on failure

NOTES
       Default comparison for unsigned: memcmp(key1, key2, size)
       Default comparison for signed: proper signed integer comparison
       Map maintains AVL balance: |height(left) - height(right)| <= 1
       All operations are O(log n) for balanced tree
       Map must be destroyed with cbmap_destroy() when done

SEE ALSO
       cbmap_create_full, cbmap_destroy

------------------------------------------------------------------------------
NAME
       cbmap_create_full - create a BST map with full customization

SYNOPSIS
       #include <cbstmap.h>

       cbmap cbmap_create_full(bool keys_are_signed_ints,
                               bool keys_are_strings,
                               ccol_memmgmt_procs_t *mmgmt_procs,
                               ccol_comparison_proc_t custom_comparison_proc,
                               char **err);

DESCRIPTION
       Creates a self-balancing BST map with full control over key signedness,
       string key support, memory management, and comparison function.

PARAMETERS
       keys_are_signed_ints     Treat integer keys as signed
       keys_are_strings         If true, use strcmp for key comparison
       mmgmt_procs              Custom memory management, or NULL
       custom_comparison_proc   Custom comparison function, or NULL
       err                      Optional pointer to receive error string

RETURN VALUE
       Returns pointer to newly created BST map on success
       Returns NULL on failure

NOTES
       Default comparison for unsigned: memcmp
       Default comparison for signed: proper signed comparison (1, 2, 4, 8 bytes)
       Default comparison for strings: strcmp (when keys_are_strings is true)
       Custom comparison receives void* pointers to key data
       All tree operations are iterative using explicit stacks

SEE ALSO
       cbmap_create, cbmap_destroy

                       CMEMPOOL.H - MEMORY POOLS

------------------------------------------------------------------------------
NAME
       mempool_create - create a fixed-size memory pool

SYNOPSIS
       #include <cmempool.h>

       mempool *mempool_create(size_t elem_count, 
                               size_t elem_size,
                               bool fallback_to_dynamic_memory, 
                               bool single_threaded,
                               ccol_memmgmt_procs_t *mmgmt_procs, 
                               char **err);

DESCRIPTION
       Creates a memory pool that manages a fixed number of fixed-size elements.
       All allocations return elements of the same size. Provides O(1) allocation
       and deallocation using a free list.

PARAMETERS
       elem_count                 Number of elements in the pool (must be > 0)
       elem_size                  Size of each element in bytes (must be > 0,
                                 minimum sizeof(uintptr_t))
       fallback_to_dynamic_memory If true, allocate from heap when pool exhausted
       single_threaded           If true, omit locking (faster but not thread-safe)
       mmgmt_procs               Custom memory management, or NULL for default
       err                       Optional pointer to receive error string

RETURN VALUE
       Returns pointer to newly created memory pool on success
       Returns NULL on failure

NOTES
       Actual element size includes header overhead
       If elem_size < sizeof(uintptr_t), it is rounded up
       Thread-safe if single_threaded is false (uses read-write locks)
       With fallback enabled, pool never fails allocation (until system OOM)
       Pool must be destroyed with mempool_destroy() when done

SEE ALSO
       mempool_create_from_preallocated_buffer, mempool_destroy, mempool_alloc_entry

------------------------------------------------------------------------------
NAME
       mempool_create_from_preallocated_buffer - create pool from user buffer

SYNOPSIS
       #include <cmempool.h>

       mempool *mempool_create_from_preallocated_buffer(
           void *buffer, 
           size_t buf_size, 
           size_t elem_size,
           bool fallback_to_dynamic_memory, 
           bool single_threaded,
           ccol_memmgmt_procs_t *mmgmt_procs, 
           char **err);

DESCRIPTION
       Creates a memory pool using a user-provided buffer instead of allocating
       from the heap. Useful for embedded systems or when heap allocation is
       undesirable.

PARAMETERS
       buffer                     Pointer to preallocated buffer
       buf_size                   Size of buffer in bytes
       elem_size                  Size of each element in bytes
       fallback_to_dynamic_memory If true, allocate from heap when pool exhausted
       single_threaded           If true, omit locking
       mmgmt_procs               Custom memory management, or NULL
       err                       Optional pointer to receive error string

RETURN VALUE
       Returns pointer to newly created memory pool on success
       Returns NULL on failure

NOTES
       Element count is calculated as buf_size / (elem_size + header_overhead)
       Buffer is not freed by mempool_destroy() (user manages buffer lifetime)
       Pool struct itself is still allocated via mmgmt_procs
       If elem_size < sizeof(uintptr_t), it is adjusted

SEE ALSO
       DECLARE_PREALLOCATED_MEMPOOL_BUFFER, mempool_create, mempool_destroy

------------------------------------------------------------------------------
NAME
       mempool_alloc_entry - allocate an entry from the pool

SYNOPSIS
       #include <cmempool.h>

       void *mempool_alloc_entry(mempool *mp);

DESCRIPTION
       Returns a pointer to an available entry from the pool. If the pool is
       exhausted and fallback is enabled, allocates from the heap. Contents are
       uninitialized.

PARAMETERS
       mp   Memory pool to allocate from

RETURN VALUE
       Returns pointer to allocated entry on success
       Returns NULL if pool exhausted and no fallback enabled

NOTES
       O(1) complexity
       Returned memory is uninitialized
       Thread-safe if pool created with single_threaded=false
       Entry must be freed with mempool_free_entry()

SEE ALSO
       mempool_calloc_entry, mempool_free_entry

------------------------------------------------------------------------------
NAME

NAME
       r_mempool_create - create a ranged memory pool

SYNOPSIS
       #include <cmempool.h>

       r_mempool *r_mempool_create(
           uint8_t smallest_size_power_of_two,
           uint8_t largest_size_power_of_two,
           uint8_t number_of_smallest_size_elems_power_of_two,
           r_memory_fallback_policy_t fb_policy,
           bool single_threaded,
           ccol_memmgmt_procs_t *mmgmt_procs,
           char **err);

DESCRIPTION
       Creates a ranged memory pool that manages multiple fixed-size pools
       organized by power-of-2 sizes. Automatically selects the appropriate
       sub-pool for each allocation request.

PARAMETERS
       smallest_size_power_of_two              Exponent for smallest pool size
       largest_size_power_of_two               Exponent for largest pool size
       number_of_smallest_size_elems_power_of_two  Exponent for smallest pool count
       fb_policy                               Fallback policy for exhausted pools
       single_threaded                         If true, omit locking
       mmgmt_procs                             Custom memory management, or NULL
       err                                     Optional pointer to receive error string

RETURN VALUE
       Returns pointer to newly created ranged memory pool on success
       Returns NULL on failure

NOTES
       Each sub-pool has half the elements of the previous size
       Minimum allowed smallest size is 16 bytes (power = 4)
       Example: smallest=4 (16 bytes), largest=6 (64 bytes), count=8 (256 elems)
         Creates pools: 16 bytes (256 elems), 32 bytes (128), 64 bytes (64)
       Fallback policies:
         - fallback_disabled: Returns NULL when pools exhausted
         - fallback_at_first_exhaustion: Uses heap when first pool exhausted
         - fallback_at_last_exhaustion: Uses heap only after all pools exhausted

SEE ALSO
       r_mempool_create_from_preallocated_buffer, r_mempool_destroy

       which of the channel's two underlying circular queues to wrap:

         ccol_select_read  -- owner <- workers_to_owner_cq  (same as chan_recv_zc)
                           worker <- owner_to_workers_cq
         ccol_select_write -- owner -> owner_to_workers_cq  (same as chan_send_zc)

       never needs access to the opaque channel struct.  The channel pointer is

       circular_queue for the calling thread and direction.

       Passing that to ccol_select() will cause it to return ccol_invalid_args.

       ccol_select() iteration.  If the calling thread's role changes between
       construction and the actual ccol_select() call, reconstruct the selectable.

       Prefer the selectable_from_chan() macro at call sites for stylistic
       consistency with selectable_from_circq() and selectable_from_dynq().

NAME
       ccol_select - wait for a message on any of several queues (blocking,
       zero-copy)

SYNOPSIS
       #include <cthreadcomm.h>

       ccol_retval_t ccol_select(c_message_t *buf, size_t *ready_index,
                               size_t n, ccol_selectable *selectables);

       ccol_retval_t ccol_select_timed(c_message_t *buf, size_t *ready_index,
                               size_t n, ccol_selectable *selectables,
                               int timeout_ms);

DESCRIPTION
       Blocks the calling thread until at least one of the n selectables has a
       message available.  When a message is found, ownership of its data pointer
       is transferred into *buf (zero-copy, exactly as individual recv functions
       do) and the 0-based index of the ready selectable is written into
       *ready_index.

       ccol_select_timed() accepts an additional timeout_ms argument.  Passing -1
       produces the same behaviour as ccol_select() (wait indefinitely).  Passing 0
       performs a non-blocking poll.  A positive value is the maximum number of
       milliseconds to wait before returning ccol_timed_out.  The deadline is
       computed once before the loop using CLOCK_MONOTONIC so that system time
       adjustments cannot extend the timeout.

       ccol_select() is equivalent to ccol_select_timed(buf, ready_index, n,
       selectables, -1).

       A ccol_selectable wraps a circular_queue, a dynamic_queue, or a raw file
       descriptor.  Build one with selectable_from_circq(), selectable_from_dynq(),
       selectable_from_chan(), or selectable_from_fd().  Channels are pre-resolved
       to their underlying circular queue before being passed here; ccol_select()
       treats them identically to plain circular queues.

       When at least one fd selectable is present, both functions create an
       epoll(7) instance internally (EPOLL_CLOEXEC) and watch both user file
       descriptors and per-waiter eventfd(2) descriptors used to bridge queue
       notifications into the epoll event loop.  User fds are registered once
       before the main retry loop with level-triggered semantics (EPOLLIN or
       EPOLLOUT) and remain registered for the entire call; closing the epoll
       instance on return removes them automatically.  Per-queue eventfds are
       created on the first registration and reused across iterations (drained
       with a non-blocking read in Phase 3 instead of closed), so subsequent
       iterations incur zero epoll_ctl or eventfd syscalls.  When no fd
       selectables are present, the lighter condvar path is used and no epoll
       instance is created.

       One waiter node per selectable is heap-allocated at the start of the call
       and freed on every return path.  There is no fixed cap on the number of
       concurrent threads calling ccol_select() on the same queue.

   Locking protocol
       Each loop iteration has three phases:

       Phase 1 (scan + register)
           For each selectable, acquire the queue's mutex.  If data is present,
           receive it immediately and return.  If the queue is empty and writing
           is enabled, prepend a heap-allocated waiter node to the queue's
           internal waiter list, then release the mutex.

       Phase 2 (wait)
           Condvar path (no fd selectables): lock the local sel_mtx and call
           pthread_cond_wait() (or pthread_cond_timedwait() when a deadline is
           set) on the local CLOCK_MONOTONIC sel_cond.  The ready flag is
           checked in a while loop to absorb spurious wakeups and signals that
           arrived between Phase 1 and the actual sleep.  Producers set the
           ready flag under sel_mtx before signalling sel_cond, so no wakeup
           is ever lost.  On ETIMEDOUT with ready still false, ccol_timed_out
           is returned.
           Epoll path (fd selectables present): call epoll_wait() with the
           remaining timeout recomputed on each attempt (including EINTR
           retries) so signal delivery cannot extend the deadline.  n_ready==0
           means the deadline elapsed; ccol_timed_out is returned.

       Phase 3 (deregister)
           Re-acquire each queue's mutex, splice the waiter node out of the
           list, and loop back to Phase 1 to find which queue has data.

       Lock ordering: producers acquire (queue mutex, then sel_mtx).  ccol_select
       holds each queue mutex alone in Phases 1 and 3, and sel_mtx alone in
       Phase 2.  No two mutexes are ever held simultaneously by ccol_select, so
       there is no lock-ordering cycle and therefore no deadlock.

       Node lifetime: producer notification happens while the producer holds the
       queue mutex.  Deregistration also requires the queue mutex.  A node
       therefore cannot be removed from the list while a producer is traversing
       it -- the heap buffer backing the node array cannot be freed mid-traversal.

PARAMETERS
       buf           Buffer to receive the message.  On ccol_success the caller
                     owns buf->data and is responsible for freeing it.
       ready_index   Pointer to a size_t that receives the 0-based index of the
                     selectable that yielded the message.  Valid only when
                     ccol_success is returned.
       n             Number of elements in the selectables array.  Must be >= 1.
       selectables   Array of n ccol_selectable values to monitor.
       timeout_ms    (ccol_select_timed only) Maximum milliseconds to wait.
                     -1: infinite wait (same as ccol_select).  0: non-blocking
                     poll.  Positive: wait up to that many milliseconds.

RETURN VALUE
       ccol_success            A selectable is ready and *ready_index is set.

                               Queue read win: *buf is populated with the
                               dequeued message (zero-copy ownership transfer);
                               caller must free buf->data.

                               fd read win: ccol_select() has already read data
                               from the fd into buf->data (heap-allocated;
                               caller must free) and set buf->size to the byte
                               count.  How much data is read per call depends on
                               fd type; see the NOTES section below.  EOF yields
                               buf->data = NULL, buf->size = 0.

                               Write-direction win (queue or fd): buf is left
                               untouched; no queue slot is reserved and no I/O
                               is performed on the fd.  Caller calls the
                               appropriate send or write(2) afterwards.

       ccol_timed_out          (ccol_select_timed only) timeout_ms elapsed
                               before any selectable became ready.  buf and
                               *ready_index are unmodified.

       ccol_invalid_args       buf, ready_index, or selectables is NULL; n is 0;
                               a queue selectable contains a NULL queue pointer;
                               an fd selectable contains a negative fd; or an
                               unrecognised type tag is present.
       ccol_not_enough_memory  malloc failed to allocate the internal waiter node
                               array (one node per selectable).
       ccol_unexpected_failure epoll_create1(2), eventfd(2), epoll_ctl(2), or
                               the internal read(2) from a ready fd failed with
                               an unexpected errno.  Only possible when at least
                               one fd selectable is present.  Treat as a fatal
                               configuration error (e.g., fd-table exhaustion or
                               memory pressure).
       ccol_msg_too_large      The data available on a read-direction fd selectable
                               exceeded the limit set in max_fd_read_bytes.  buf
                               is left untouched.  *ready_index IS set to the
                               index of the offending fd selectable so the caller
                               can identify it.  Note: ccol_select() may have
                               already consumed some or all bytes from the fd's
                               kernel buffer during the read attempt; for datagram
                               fds the entire datagram is consumed (lost).  The
                               caller should not assume unread bytes remain.

NOTES
       Waiter nodes are heap-allocated (one per selectable) for the duration of
       the call and freed on every return path, including error paths.  This
       eliminates the VLA stack-overflow risk for large n.

       Any number of threads may simultaneously call ccol_select() watching the
       same queue.  The waiter list is a dynamically-linked list with no fixed
       capacity.

       When a message arrives in a queue watched by multiple ccol_select()
       threads, only the head waiter is woken (single-wake).  If the queue still
       has more messages after the first waiter claims one, it immediately wakes
       the next waiter (cascading wake).  This prevents both thundering herd and
       liveness gaps when multiple messages arrive in quick succession.

       When writing is disabled on a queue (via circq_disable_sending() or
       dynmq_disable_sending()), read-direction waiters on that queue remain
       registered and continue waiting.  ccol_select() will unblock once sending
       is re-enabled and a message arrives.  Write-direction waiters are woken
       immediately so they can observe the new state.

       File descriptors must be kept open for the entire duration of the
       ccol_select() call.  Closing an fd that ccol_select() is currently
       watching produces undefined behaviour.

       For read-direction fd wins, ccol_select() performs the read internally.
       To bound the heap allocation, set max_fd_read_bytes on the selectable
       (via selectable_from_fd_limited()).  getsockopt(SO_TYPE) is used to detect
       SOCK_DGRAM and SOCK_SEQPACKET fds:

       Datagram fds (SOCK_DGRAM, SOCK_SEQPACKET): a single read(2) into a 66 KiB
       buffer captures the full datagram without truncation for any standard
       IPv4/IPv6 UDP payload (max 65,507 bytes).  The grow-retry loop is suppressed
       regardless of O_NONBLOCK.  When max_fd_read_bytes is set and the datagram
       size exceeds it, ccol_msg_too_large is returned.

       Stream and non-socket fds (pipes, timerfd, signalfd, etc.): a 4 KiB initial
       buffer is used.  If the fd has O_NONBLOCK, the loop grows the buffer (up to
       max_fd_read_bytes + 1 when a limit is set) until EAGAIN.  If blocking,
       exactly one read(2) call is made using a buffer of
       max(4096, max_fd_read_bytes) bytes; passing max_fd_read_bytes > 4096 via
       selectable_from_fd_limited() therefore allows reading up to
       max_fd_read_bytes bytes in a single blocking call.  Any data remaining in
       the fd buffer after that single read will be returned on the next
       ccol_select() call because epoll(7) is level-triggered.  Use O_NONBLOCK on
       the fd if full-drain behaviour per call is required.  The fd's flags are
       inspected read-only via fcntl(F_GETFL) and are never modified, so concurrent
       users of the same fd are unaffected.

       Uses POSIX mutexes and condition variables internally; epoll(7) and
       eventfd(2) when fd selectables are present (Linux only).

EXAMPLES
       /* Fan-in from two circular queues and a channel: */
       c_message_t msg;
       size_t idx;
       ccol_retval_t r = ccol_select(&msg, &idx, 3,
           (ccol_selectable[]){
               selectable_from_circq(q0, ccol_select_read),
               selectable_from_circq(q1, ccol_select_read),
               selectable_from_chan(ch, ccol_select_read),
           });
       if (r == ccol_success) {
           printf("got message from selectable %zu\n", idx);
           free(msg.data);
       }

       /* Equivalent using the convenience macro: */
       r = ccol_select_va(&msg, &idx,
               selectable_from_circq(q0, ccol_select_read),
               selectable_from_dynq(dq, ccol_select_read),
               selectable_from_chan(ch, ccol_select_read));

       /* Mixed read+write: wake when q0 has room OR dq has a message: */
       r = ccol_select_va(&msg, &idx,
               selectable_from_circq(q0, ccol_select_write),
               selectable_from_dynq(dq,  ccol_select_read));

       /* Fan-in from a queue and a pipe read end: */
       int pfd[2];
       pipe(pfd);
       r = ccol_select_va(&msg, &idx,
               selectable_from_circq(q0, ccol_select_read),
               selectable_from_fd(pfd[0], ccol_select_read));
       if (r == ccol_success) {
           if (idx == 0) {
               /* queue won -- buf has the dequeued message */
               printf("queue message size: %zu\n", msg.size);
               free(msg.data);
           } else {
               /* fd won -- ccol_select already read the data into buf */
               printf("read %zu bytes from fd\n", msg.size);
               free(msg.data);   /* msg.data may be NULL on EOF */
           }
       }

       /* Timed wait with 500 ms deadline: */
       r = ccol_select_timed_va(&msg, &idx, 500,
               selectable_from_circq(q0, ccol_select_read));
       if (r == ccol_timed_out) {
           printf("no message within 500 ms\n");
       }

       /* Bounded fd read: reject messages larger than 64 KiB: */
       r = ccol_select_va(&msg, &idx,
               selectable_from_fd_limited(pfd[0], ccol_select_read, 65536));
       if (r == ccol_msg_too_large) {
           printf("oversized message rejected\n");
       }

------------------------------------------------------------------------------
NAME
       clrucache_create_full - create an LRU cache with full control

SYNOPSIS
       #include <clrucache.h>

       clru_cache clrucache_create_full(size_t capacity,
                                        ccol_data_type key_type,
                                        ccol_data_type val_type,
                                        clru_remote_getter_t getter,
                                        clru_remote_setter_t setter,
                                        clru_eviction_cb_t eviction_cb,
                                        ccol_memmgmt_procs_t *mprocs,
                                        char **err);

DESCRIPTION
       Creates a thread-safe LRU cache backed by a hash map (O(1) lookup) and
       a doubly-linked list (O(1) eviction). The cache holds up to capacity
       live entries; inserting beyond that limit evicts the least-recently-used
       entry first.

       This is the low-level constructor. Prefer the clru_construct() macro
       family in application code.

PARAMETERS
       capacity     Maximum number of live entries before eviction occurs.
                    Must be > 0.
       key_type     ccol_data_type of keys, as returned by
                    determine_ccol_data_type().
       val_type     ccol_data_type of values (used for type tracking only;
                    the internal map always stores heap pointers).
       getter       Remote getter callback, or NULL. Called on cache miss to
                    fetch the value from an external source. On success it
                    must heap-allocate the value data, assign it to val->ptr,
                    set val->size to the byte length, and return true. The
                    cache takes ownership of val->ptr and frees it. On
                    failure it must leave val->ptr as NULL and return false.
       setter       Remote setter callback, or NULL. Called synchronously
                    before each cache update. Receives the key and value as
                    const cmap_pair* pointers. If it returns false the cache
                    is not updated.
       eviction_cb  Eviction callback, or NULL. Called while the cache mutex
                    is held when an entry is evicted to make room. Receives
                    the evicted key and value as const cmap_pair* pointers.
                    Must not call back into the cache.
       mprocs       Custom memory management procedures, or NULL for default
                    malloc/free.
       err          Optional pointer to receive an error string on failure.

RETURN VALUE
       Returns a new cache handle on success.
       Returns NULL on failure; *err is set if err is non-NULL.

NOTES
       Thread-safe: all operations on the returned handle are protected by
       an internal mutex and per-entry condition variables.
       The eviction callback is invoked while the cache mutex is held -- it
       must not re-enter the cache.
       The destroy function must be called only after all other threads have
       finished using the cache.

SEE ALSO
       clru_construct, clru_init, __clrucache_destroy, clrucache_get_full,
       clrucache_set_full

------------------------------------------------------------------------------
NAME
       __clrucache_destroy - destroy an LRU cache (internal)

SYNOPSIS
       #include <clrucache.h>

       void __clrucache_destroy(clru_cache cache);

DESCRIPTION
       Evicts all remaining live entries (invoking the eviction callback for
       each while the mutex is held), destroys the internal hash map, frees
       the mutex, and releases all memory.

       This is the low-level destructor. In application code use the
       clru_destroy() macro, which also sets the pointer to NULL.

PARAMETERS
       cache  Cache handle to destroy. Passing NULL is a no-op.

WARNINGS
       Must be called only after all other threads have stopped using the
       cache. Outstanding blocked get or set calls will corrupt memory if
       __clrucache_destroy is called while they are in progress.

SEE ALSO
       clru_destroy, clrucache_create_full

------------------------------------------------------------------------------
NAME
       clrucache_get_full - retrieve a value by key

SYNOPSIS
       #include <clrucache.h>

       ccol_retval_t clrucache_get_full(clru_cache cache,
                                        const cmap_pair *key_pair,
                                        cmap_pair *val_out);

DESCRIPTION
       Looks up key_pair in the cache and, on success, heap-allocates a copy
       of the associated value using the cache's custom allocator (if one was
       provided at construction) or malloc() otherwise, assigns the pointer to
       val_out->ptr, and sets val_out->size to the stored byte size. The caller
       is responsible for freeing val_out->ptr with the matching allocator: the
       custom free function if a custom allocator was configured, or free()
       otherwise.

       Cache hit: the entry is moved to the most-recently-used position and
       a heap-allocated copy of its value is written into val_out.

       Cache miss without a remote getter: returns ccol_key_not_found.

       Cache miss with a remote getter: a placeholder entry is inserted and
       the remote getter is called outside the mutex. Concurrent get calls
       for the same key coalesce: only the first caller runs the fetch; all
       others block until the result is available. On getter success the
       fetched value is cached and a heap-allocated copy is written into
       val_out for all waiters. On getter failure, all waiters receive
       ccol_key_not_found.

       If a set is in progress for the same key this call blocks until the
       set completes and the cache is in a consistent state.

PARAMETERS
       cache     Cache handle.
       key_pair  Key to look up (ptr and size fields must be set).
       val_out   Output pair populated on success. val_out->ptr is set to a
                 heap-allocated copy of the value (caller must free it with
                 the cache's custom allocator, or free() if no custom
                 allocator was provided); val_out->size is set to the stored
                 value size. Must not be NULL.

RETURN VALUE
       ccol_success           val_out populated; caller owns val_out->ptr.
       ccol_key_not_found     Key absent and no getter, or getter failed.
       ccol_invalid_args      cache, key_pair, or val_out is NULL.
       ccol_not_enough_memory Allocation failed (placeholder entry or output
                              copy).

NOTES
       The returned val_out->ptr is allocated with the cache's custom
       allocator if one was provided at construction, or with the standard
       malloc() otherwise. Free it with the matching allocator.
       When the value type is char*, the returned copy is a heap-allocated
       null-terminated string: this allocation is also made with the custom
       allocator (or malloc() if none was configured).
       The cache never exposes a pointer into its internal storage (such a
       pointer would dangle after eviction).
       O(1) average-case complexity for the hash map lookup.

SEE ALSO
       clru_get, clrucache_set_full, clrucache_create_full

NAME
       clrucache_set_full - store a value by key (write-through)

SYNOPSIS

                                        const cmap_pair *key_pair,

       ccol_unexpected_failure  Remote setter returned false; cache unchanged.
       ccol_invalid_args        cache, key_pair, or val_pair is NULL; or
                                val_pair->size is 0.
       ccol_not_enough_memory   Memory allocation for the new value failed.

NOTES
       O(1) average-case complexity for the hash map operations.

       clru_set, clrucache_get_full, clrucache_create_full

------------------------------------------------------------------------------
NAME
       clog_open_file_mp - create a file-backed logger with optional log rotation

SYNOPSIS
       #include <clogger.h>

       clog clog_open_file_mp(const char *path, clog_level_t min_level,
                           const clog_rotation_cfg_t *cfg,
                           ccol_memmgmt_procs_t *mprocs);

DESCRIPTION
       Creates a new logger that writes to the file at path. The file is
       created if it does not exist (mode 0644) and opened for appending if
       it does. When the file already exists its current size is read via
       fstat(2) so that size-based rotation thresholds are applied correctly
       even when appending to a pre-existing log.

       Log rotation is enabled only when cfg is non-NULL. Both size-based
       and time-based rotation may be enabled independently:

       Size rotation: triggers after the write that causes the cumulative
       bytes written to reach or exceed cfg->max_file_size. Checked after
       each write.

       Time rotation: triggers on the first write after
       cfg->rotation_interval_secs seconds have elapsed since the last
       rotation (or since the logger was created). Checked before each write.

       When a rotation fires, the current file is atomically renamed to
       <path>.<YYYYMMDDHHMMSS>. If a file with that name already exists a
       collision suffix _1, _2, ... (up to _9999) is appended. A new empty
       log file is then opened. When cfg->max_rotated_files > 0, the oldest
       rotated files beyond the limit are deleted automatically.

       When cfg->compress_rotated is true, each rotated file is gzip-
       compressed via zlib immediately after rotation. The compressed file
       is written to <rotated-name>.gz and the original uncompressed file
       is removed on success. Compression runs outside the logger mutex so
       log writers are never stalled while the I/O completes. The resulting
       .gz files are compatible with standard gunzip(1) and zcat(1) tools.
       Requires linking the application with -lz.

PARAMETERS
       path       Destination log file path. Copied internally.
       min_level  Messages below this level are silently dropped.
       cfg        Rotation configuration, or NULL to disable rotation.
                  Fields with value 0 fall back to their respective
                  CLOG_DEFAULT_* constant:
                    size_rotation_enabled  - enable size-based rotation
                    max_file_size          - bytes threshold (0 = 10 MiB)
                    time_rotation_enabled  - enable time-based rotation
                    rotation_interval_secs - seconds (0 = 86400 = 1 day)
                    max_rotated_files      - files kept; 0 = no limit
                    compress_rotated       - gzip-compress each rotated file
       mprocs     Custom allocator (malloc/free/calloc/realloc function
                  pointers), or NULL to use the default system allocator.
                  All four pointers must be set when mprocs is non-NULL;
                  a partially-populated struct causes the call to return
                  NULL without creating a logger. The struct is copied
                  internally; the caller may free it after this call.
                  The allocator is inherited by all loggers derived from
                  this one via clog_derive().

RETURN VALUE
       Returns a new logger handle on success.
       Returns NULL on failure (bad path, permission error, invalid mprocs,
       allocation failure, etc.).

NOTES
       The logger owns the file descriptor and closes it on clog_close().
       O_CLOEXEC is set on the file descriptor.
       The rename and new-fd open are performed within the locked section
       so no log lines are lost or interleaved across the rotation boundary.
       When compress_rotated is true, the gzip compression step runs after
       the new fd is already in use, outside the mutex, to avoid stalling
       concurrent log writers during the compression I/O.

SEE ALSO
       clog_open_fd_mp, clog_close, CLOG_DEFAULT_MAX_FILE_SIZE,
       CLOG_DEFAULT_ROTATION_INTERVAL, CLOG_DEFAULT_MAX_ROTATED_FILES

NAME
       clog_close - flush, close, and free a logger

SYNOPSIS
       #include <clogger.h>

       void clog_close(clog logger);

DESCRIPTION
       Releases all resources owned by the logger:
         - If the logger is file-backed and this is the last handle sharing

           closed), the file descriptor is closed.  If derived loggers still
           exist the fd remains open and the file's ref-count is decremented.

       log_* macros on the same handle. Callers are responsible for ensuring
       all logging threads have finished before calling clog_close().

NAME
       clog_set_field - attach or update a persistent structured field

SYNOPSIS
       #include <clogger.h>

       void clog_set_field(clog logger, const char *key, const char *value);

DESCRIPTION
       Associates key with value as a persistent field. The field appears in
       every subsequent log line emitted by this logger, inserted between the
       func= and msg= tokens:

         ts=... level=INFO proc=prog(pid):tname(tid) src=... func=... key=value msg=...

       Both key and value are copied internally; the caller may free or
       modify them after this call without affecting the logger.

       If key already exists its value is replaced. Otherwise a new field is
       prepended to the field list.

       Values that contain spaces, '=', '"', '\\', or ASCII control
       characters are automatically double-quoted and backslash-escaped in
       the output:
         - '"' -> \"
         - '\\' -> \\
         - control chars -> \xNN

       logger  Logger handle.
       key     Field name. Copied internally. Should not contain '=' or
               whitespace (no enforcement, but the resulting logfmt line
               would be malformed).
       value   Field value. Copied internally. May be arbitrary text.

NOTES
       Thread-safe: protected by the per-logger mutex.
       The field list is a linked list. For large field sets, iteration is
       O(n) per write; prefer a small number of long-lived fields.

       clog_remove_field, clog_clear_fields

NAME
       clog_derive - create a derived logger sharing the same output target

SYNOPSIS
       #include <clogger.h>

       clog clog_derive(clog parent);

DESCRIPTION
       Creates a new derived logger that writes to the same file descriptor as
       parent and shares the same synchronisation mutex and log-rotation state.
       All writes from parent and every derived logger are therefore serialised
       by the same lock and land in the same destination.

       At the moment of the call, the derived logger receives a snapshot of
       parent's structured fields and minimum log level.  After creation the
       two loggers maintain fully independent field maps and level settings:
       adding, removing, or changing fields or the level on one logger has no
       effect on the other.

       Log rotation (if configured on the original file-backed logger) continues
       to be managed by the root logger; derived loggers benefit transparently.

       Release a derived logger with clog_close() when it is no longer needed.
       The underlying file is kept open until all handles (root and every derived
       logger) have been closed -- the shared backing store uses reference counting.

PARAMETERS
       parent  Source logger to derive from.

RETURN VALUE
       Returns a new derived logger handle on success.
       Returns NULL if memory allocation fails.

NOTES
       Derived loggers inherit the output format and syslog facility from the
       shared backing store; changes to those settings via any handle are
       immediately visible to all handles writing to the same fd.
       Thread-safe: the field snapshot is taken under the shared mutex.

SEE ALSO
       clog_open_fd_mp, clog_open_file_mp, clog_close

------------------------------------------------------------------------------
NAME
       clog_set_format - change the output format at runtime

SYNOPSIS
       #include <clogger.h>

       void clog_set_format(clog logger, clog_format_t fmt);

DESCRIPTION
       Updates the output format of the shared backing store.  Because the
       format is stored on the shared store rather than the per-logger struct,
       the change is immediately visible to every logger handle that writes to
       the same file descriptor (the root logger and all derived loggers).

       Three formats are available:

         CLOG_FMT_LOGFMT (0)
           Default.  Every record is emitted as a single logfmt line:
             ts=<ISO-8601-UTC> level=<L> proc=<name>(<pid>):<tname>(<tid>)
             src=<file>:<line> func=<fn> [fields] msg=<text>
           log_error, log_alert, and log_fatal append tab-indented backtrace
           continuation lines after the main record.

         CLOG_FMT_JSON (1)
           Every record is emitted as a single NDJSON object followed by a
           newline:
             {"ts":"...","level":"...","proc":"name(pid):tname(tid)",
              "src":"file:N","func":"...", [fields,] "msg":"..."}
           Structured fields become top-level JSON keys.  For log_error,
           log_alert, and log_fatal the backtrace is embedded as a "bt"
           string array inside the same JSON object.

         CLOG_FMT_SYSLOG (2)
           RFC 5424 syslog format.  Available only for fd-based loggers
           (clog_open_fd / clog_open_fd_mp).  Calling clog_set_format() with
           CLOG_FMT_SYSLOG on a file-backed logger is a silent no-op; the
           format remains unchanged.

           Each record is emitted as a single line:
             <PRI>1 TIMESTAMP HOSTNAME APP-NAME PID MSGID [ccol ...] MSG\n

             PRI        = (facility * 8) + RFC 5424 severity (mapped from
                          clog_level_t: TRACE=7, DEBUG=7, INFO=6, WARN=4,
                          ERROR=3, ALERT=1, FATAL=0)
             TIMESTAMP  = ISO 8601 UTC (same format as logfmt/JSON)
             HOSTNAME   = result of gethostname(2), or "-" if unavailable
             APP-NAME   = process name from /proc/self/comm or argv[0], or "-"
             PID        = getpid(2)
             MSGID      = level string ("INFO", "ERROR", etc.)
             [ccol ...]  = RFC 5424 SD-ELEMENT with SD-ID "ccol" carrying
                          proc="[<name>(<pid>):<tname>(<tid>)\]",
                          src="<file>:<line>", func="<fn>", and any fields
                          attached via clog_set_field()
             MSG        = the formatted message text

           The facility is controlled by clog_set_facility() and defaults to
           CLOG_SYSLOG_USER.  Backtraces (log_error, log_alert, log_fatal) are
           emitted as separate syslog messages, one per frame, using the same
           PRI and MSGID as the original record.

           The fd must be connected to a syslog daemon before calling
           clog_set_format().  On Linux this is typically a SOCK_DGRAM Unix
           socket at /dev/log.  Keep individual messages under 2 KiB to
           stay within typical syslogd datagram limits.

PARAMETERS
       logger  Logger handle.

NAME
       clog_set_facility - change the syslog facility

SYNOPSIS
       #include <clogger.h>

       void clog_set_facility(clog logger, clog_syslog_facility_t facility);

DESCRIPTION
       Updates the syslog facility stored on the shared backing store.  The
       facility is encoded into the PRI field of every RFC 5424 message as
       (facility * 8) + severity.  Because the value lives on the shared store
       the change is visible to every logger handle that writes to the same file
       descriptor (the root logger and all derived loggers).

       This function is only meaningful when the output format is
       CLOG_FMT_SYSLOG; calling it with any other format is harmless (the
       stored value is updated but has no effect on the output until
       CLOG_FMT_SYSLOG is selected).

       Available facility codes (RFC 5424 / RFC 3164):
         CLOG_SYSLOG_KERN     (0)  -- Kernel messages
         CLOG_SYSLOG_USER     (1)  -- User-level messages (default)
         CLOG_SYSLOG_MAIL     (2)  -- Mail system
         CLOG_SYSLOG_DAEMON   (3)  -- System daemons
         CLOG_SYSLOG_AUTH     (4)  -- Security / authorization messages
         CLOG_SYSLOG_SYSLOG   (5)  -- Internal syslogd messages
         CLOG_SYSLOG_LPR      (6)  -- Line printer subsystem
         CLOG_SYSLOG_NEWS     (7)  -- Network news subsystem
         CLOG_SYSLOG_UUCP     (8)  -- UUCP subsystem
         CLOG_SYSLOG_CRON     (9)  -- Clock daemon
         CLOG_SYSLOG_AUTHPRIV (10) -- Security / authorization (private)
         CLOG_SYSLOG_FTP      (11) -- FTP daemon
         CLOG_SYSLOG_LOCAL0   (16) -- Local use 0
         ...
         CLOG_SYSLOG_LOCAL7   (23) -- Local use 7

PARAMETERS
       logger    Logger handle.
       facility  Desired facility code (clog_syslog_facility_t).

NOTES
       Thread-safe: protected by the shared backing-store mutex.
       The default facility is CLOG_SYSLOG_USER.

SEE ALSO
       clog_get_facility, clog_set_format

NAME
       _clog_write - internal log-line writer (use the log_* macros instead)

SYNOPSIS
       #include <clogger.h>

       void _clog_write(clog logger, clog_level_t level,
                        const char *file, int line, const char *func,
                        bool with_backtrace, const char *fmt, ...)
           __attribute__((format(printf, 7, 8)));

DESCRIPTION
       Formats and emits a single structured log line. This is the function
       that all log_* macros expand to; application code should call the
       macros rather than this function directly.

       The write sequence (all steps under the per-logger mutex) is:

         1. Check level >= min_level. If filtered AND level != CLOG_FATAL,
            return immediately (no output). CLOG_FATAL bypasses this check.
         2. Format the user message with vsnprintf into a 1024-byte stack
            buffer; spill to heap if the formatted text exceeds 1024 bytes.
         3. If time rotation is enabled, check and fire if due.
         4. Build the log record according to the shared backing-store format:

            CLOG_FMT_LOGFMT:
              ts=<ISO-8601-UTC> level=<L> proc=<name>(<pid>):<tname>(<tid>)
              src=<file>:<line> func=<func> [key=value ...] msg=<message>

            CLOG_FMT_JSON:
              {"ts":"...","level":"...","proc":"name(pid):tname(tid)",
               "src":"file:N","func":"...", [fields,] "msg":"..." [,"bt":[...]]}
              The "bt" array is appended inline when with_backtrace is true.

         5. Write the record atomically with write(2) via an EINTR-retry loop.
         6. If with_backtrace is true, emit the backtrace:
            CLOG_FMT_LOGFMT: call backtrace() + backtrace_symbols() and
              write each frame as a tab-indented continuation line:
                \t#N symbol\n
            CLOG_FMT_SYSLOG: emit each frame as a separate RFC 5424 syslog
              message (same PRI and MSGID as the original record).
            CLOG_FMT_JSON: backtrace was already embedded as "bt" in step 4.
         7. If size rotation is enabled, check and fire if due.
         8. If level == CLOG_FATAL, call exit(EXIT_FAILURE).

       The write buffer is a per-logger dynamic buffer that starts at 4 KiB
       and grows (doubling) up to 16 MiB. It is reused across calls.

PARAMETERS
       logger          Logger handle.
       level           Severity level of this message.
       file            Source file name (__FILE__).
       line            Source line number (__LINE__).
       func            Function name (__func__).
       with_backtrace  If true, append a stack backtrace after the log line.
       fmt             printf-style format string.
       ...             Format arguments.

NOTES
       Backtrace support requires a platform with execinfo.h (glibc, macOS,
       FreeBSD) and the binary linked with -rdynamic for symbol resolution.
       On other platforms the backtrace is silently omitted.
       The format attribute causes the compiler to type-check fmt and the
       variadic arguments at compile time.

SEE ALSO
       log_trace, log_debug, log_info, log_warn, log_error, log_fatal

NAME
       cjson_parse_mp, cjson_parse - parse a null-terminated JSON string into
       a DOM tree

SYNOPSIS
       #include <cjson.h>

       cjson cjson_parse_mp(const char *json_str, char **err_str,
                            ccol_memmgmt_procs_t *mp);
       /* convenience wrapper -- default allocator */
       static inline cjson cjson_parse(const char *json_str,
                                       char **err_str);

DESCRIPTION
       Parses json_str using a recursive-descent parser and returns the root
       node of the resulting DOM tree.  All seven JSON value kinds are
       supported: null, boolean, integer (stored as long long), number (double),
       string (UTF-8, owned heap copy), array (cvec of child cjson), and object
       (chmap of char*->cjson).

       Integer literals with no decimal point or exponent are stored as
       CJSON_INTEGER; all others (including integers that overflow long long)
       as CJSON_FLOAT.  Number literals are limited to 359 decimal characters;
       longer inputs produce a parse error.  Numbers that overflow IEEE 754
       double (i.e. strtod returns infinity) also produce a parse error.

       String escape sequences \", \\, \/, \b, \f, \n, \r, \t, and
       \uXXXX (including surrogate pairs) are decoded to UTF-8.  The
       \u0000 is explicitly rejected with a parse error: cjson
       stores strings as null-terminated char * values and cannot
       represent embedded null bytes.

       cjson_parse() is a static inline wrapper that calls
       cjson_parse_mp(json_str, err_str, NULL).

PARAMETERS
       json_str  Null-terminated JSON text.
       err_str   Out-parameter for the parse error message.  On success
                 *err_str is set to NULL.  On failure a heap-allocated
                 null-terminated string is stored in *err_str; the caller
                 must free() it when done.  Pass NULL to discard the error
                 message.
       mp        Custom memory management procedures for all nodes in the
                 resulting tree, or NULL for the default allocator.

RETURN VALUE
       Root cjson node on success.
       NULL on parse failure; if err_str was non-NULL, *err_str holds the
       reason (caller must free it).

NOTES
       The caller owns the returned tree and must free it with cjson_destroy().
       The error string (when requested) is always allocated with plain
       malloc so the caller can always free it with plain free(), regardless
       of which mp was used for node allocation.

EXAMPLE
       char *err = NULL;
       cjson doc = cjson_parse_mp(input, &err, NULL);
       if (!doc) {
           fprintf(stderr, "parse failed: %s\n", err);
           free(err);
           return -1;
       }
       /* use doc ... */
       cjson_destroy(doc);

SEE ALSO
       cjson_parse_n_mp, cjson_destroy, cjson_serialize

NAME
       cjson_create_null_mp, cjson_create_bool_mp, cjson_create_int_mp,
       cjson_create_double_mp, cjson_create_string_mp, cjson_create_list_mp,
       cjson_create_dictionary_mp - allocate DOM nodes with custom allocator

       cjson_create_null, cjson_create_bool, cjson_create_int,
       cjson_create_double, cjson_create_string, cjson_create_list,
       cjson_create_dictionary - allocate DOM nodes (default allocator wrappers)

SYNOPSIS
       #include <cjson.h>

       cjson cjson_create_null_mp(ccol_memmgmt_procs_t *mp);
       cjson cjson_create_bool_mp(bool val, ccol_memmgmt_procs_t *mp);
       cjson cjson_create_int_mp(long long val, ccol_memmgmt_procs_t *mp);
       cjson cjson_create_double_mp(double val, ccol_memmgmt_procs_t *mp);
       cjson cjson_create_string_mp(const char *val, ccol_memmgmt_procs_t *mp);
       cjson cjson_create_list_mp(ccol_memmgmt_procs_t *mp);
       cjson cjson_create_dictionary_mp(ccol_memmgmt_procs_t *mp);

       /* convenience wrappers -- default allocator (NULL mp) */
       static inline cjson cjson_create_null(void);
       static inline cjson cjson_create_bool(bool val);
       static inline cjson cjson_create_int(long long val);
       static inline cjson cjson_create_double(double val);
       static inline cjson cjson_create_string(const char *val);
       static inline cjson cjson_create_list(void);
       static inline cjson cjson_create_dictionary(void);

DESCRIPTION
       Each _mp function allocates a new DOM node of the corresponding JSON
       type using mp as the allocator (NULL = default malloc/free/calloc/realloc).
       The allocator is stored per-node and used for all subsequent operations
       on that node, including destruction.

       The non-_mp variants are static inline wrappers that call the _mp
       function with NULL as the allocator.

       cjson_create_string_mp() makes a private heap copy of val.  Passing NULL
       for val produces a CJSON_NULL node (not a CJSON_STRING node with a null
       pointer), because a null C string pointer naturally maps to JSON null.
       cjson_create_list_mp() returns an empty array; children are added with
       cjson_list_push().
       cjson_create_dictionary_mp() returns an empty object; children are added with
       cjson_dictionary_set().

       cjson_create_double_mp() rejects non-finite values (Inf, -Inf, NaN) and
       returns NULL for them.  JSON has no representation for non-finite
       doubles; the parser likewise rejects numbers that overflow to infinity.
       Use cjson_set() to write a double field; it propagates the same
       validation and returns ccol_invalid_args on failure.

PARAMETERS
       mp   Custom memory management procedures for this node and its owned
            resources, or NULL for the default allocator.

RETURN VALUE
       New cjson node on success, NULL on allocation failure or (for
       cjson_create_double_mp) if val is non-finite.
       cjson_create_string_mp(NULL, mp) returns a CJSON_NULL node (never NULL
       itself, unless OOM).

NOTES
       The caller owns the returned node and must free it with cjson_destroy().
       cjson_destroy() uses each node's stored allocator -- no mp parameter needed.

SEE ALSO
       cjson_destroy, cjson_list_push, cjson_dictionary_set

NAME
       _cjson_delete - path-based delete back-end (prefer cjson_delete macro)

SYNOPSIS
       #include <cjson.h>

       ccol_retval_t _cjson_delete(cjson root, const char *path);

DESCRIPTION
       Navigates to the parent of the node addressed by the dot-separated path
       and removes and deep-frees that node.  The path uses the same syntax as
       _cjson_get: plain components address dictionary keys, '#N' components
       address list elements by zero-based index, and '\.' / '\\' are escape
       sequences for a literal dot or backslash in a key name.

       The split is performed on the LAST unescaped dot, so "a.b.c" navigates
       to parent "a.b" and removes key "c".  When path has no dot, root itself
       is the parent.

       Prefer the cjson_delete() macro for all call sites.

RETURN VALUE
       ccol_success           - node removed and freed.
       ccol_invalid_args      - root or path is NULL, path is empty, or the
                                leaf component is not valid for the parent type
                                (e.g. a non-'#N' component on a list parent).
       ccol_key_not_found     - any path component, including the parent path,
                                is absent.
       ccol_not_enough_memory - internal strdup failed.

SEE ALSO
       cjson_delete (macro), _cjson_get, _cjson_set_typed

NAME
       cyaml_parse_mp, cyaml_parse - parse a null-terminated YAML string into
       a DOM tree

SYNOPSIS
       #include <cyaml.h>

       cyaml cyaml_parse_mp(const char *yaml_str, char **err_str,
                            ccol_memmgmt_procs_t *mp);
       cyaml cyaml_parse(const char *yaml_str, char **err_str);

DESCRIPTION
       Parse a null-terminated YAML string and return the root DOM node.
       Comments (#) are silently discarded.  Tags (!tag, !!tag) are silently
       ignored.  Anchors (&name) and aliases (*name) are resolved to
       independent deep clones.  Implicit YAML 1.2 core schema typing is
       applied to unquoted scalars.

       SINGLE-DOCUMENT STREAMS
       When the input contains exactly one document (with or without a leading
       "---" marker and an optional trailing "..." marker), the root node of
       that document is returned directly.

       MULTI-DOCUMENT STREAMS
       When the input contains multiple documents separated by "---" markers,
       all document roots are collected and returned as a single CYAML_LIST
       node whose elements are the individual document roots in order.  The
       anchor table is reset between documents so an anchor defined in one
       document cannot be referenced by a later document.  After the first
       document, each subsequent document must begin with a "---" marker;
       bare content following a document end ("...") is rejected as a parse
       error.

       cyaml_parse() is an inline wrapper that passes NULL for mp.

       On parse failure *err_str (if non-NULL) receives a heap-allocated,
       null-terminated error message that the caller must free().  On success
       *err_str is set to NULL.  Pass NULL for err_str to discard error
       details.

PARAMETERS
       yaml_str   Null-terminated YAML input.
       err_str    Out-parameter for the error string, or NULL to ignore.
       mp         Custom allocator for all nodes in the resulting tree, or
                  NULL for the default allocator.

RETURN VALUE
       For a single-document input: the root cyaml node.
       For a multi-document input: a CYAML_LIST node wrapping all document
       roots.
       NULL on parse failure or OOM.

NOTES
       The caller owns the returned tree and must free it with cyaml_destroy().

       To distinguish single- from multi-document results, check the type of
       the returned node with cyaml_type().

SEE ALSO
       cyaml_parse_n_mp, cyaml_destroy, cyaml_serialize

NAME
       _cyaml_get - path-navigation back-end (prefer cyaml_get macro)

SYNOPSIS
       #include <cyaml.h>

       cyaml _cyaml_get(cyaml root, const char *path);

DESCRIPTION
       Navigate from root through a dot-separated path and return the node at
       the destination.  Path components beginning with '#' followed by
       decimal digits index into a CYAML_LIST by position; all other

         \\   -> literal '\' in the key

       _cyaml_get() directly only when the macro wrapper is inconvenient.

       node type at that step is incompatible with the operation.

NAME
       _cyaml_set_typed - path-write back-end (prefer cyaml_set macro)

SYNOPSIS
       #include <cyaml.h>

       ccol_retval_t _cyaml_set_typed(cyaml root, const char *path,
                                      cyaml_node_type_t type, void *raw,
                                      size_t raw_size, bool is_signed,
                                      bool raw_is_char_array);

DESCRIPTION
       Write a scalar value to the leaf addressed by path.  The leaf is
       created if absent (its immediate parent must already exist).  If the
       leaf exists its type is replaced unconditionally; any composite
       (list, dictionary) resources it previously held are deep-freed.

       raw is a void pointer to the C value; raw_size is sizeof the original
       C expression.  For string values, raw points to a const char * variable
       unless raw_is_char_array is true, in which case raw points directly at

         \.   -> literal '.' in the key (not a path separator)
         \\   -> literal '\' in the key
       A '\' before any other character is left unchanged.

       The cyaml_set() macro is the preferred public interface and fills in
       type, raw_size, is_signed, and raw_is_char_array automatically from the
       compile-time type of the value expression.

PARAMETERS
       root             Root of the DOM tree.
       path             Dot-separated path string.
       type             YAML type of the new value.
       raw              Pointer to the raw C value.
       raw_size         sizeof() of the original C expression.
       is_signed        Whether the integer source type is signed.
       raw_is_char_array  true when raw points directly at a char[] array.

RETURN VALUE
       ccol_success, ccol_invalid_args, ccol_key_not_found, or
       ccol_not_enough_memory.

SEE ALSO
       cyaml_set (macro), _cyaml_get

NAME
       _cyaml_delete - path-based delete back-end (prefer cyaml_delete macro)

SYNOPSIS
       #include <cyaml.h>

       ccol_retval_t _cyaml_delete(cyaml root, const char *path);

DESCRIPTION
       Navigates to the parent of the node addressed by the dot-separated path
       and removes and deep-frees that node.  The path uses the same syntax as
       _cyaml_get: plain components address dictionary keys, '#N' components
       address list elements by zero-based index, and '\.' / '\\' are escape
       sequences for a literal dot or backslash in a key name.

       The split is performed on the LAST unescaped dot, so "a.b.c" navigates
       to parent "a.b" and removes key "c".  When path has no dot, root itself
       is the parent.

       Prefer the cyaml_delete() macro for all call sites.

RETURN VALUE
       ccol_success           - node removed and freed.
       ccol_invalid_args      - root or path is NULL, path is empty, or the
                                leaf component is not valid for the parent type
                                (e.g. a non-'#N' component on a list parent).
       ccol_key_not_found     - any path component, including the parent path,
                                is absent.
       ccol_not_enough_memory - internal strdup failed.

SEE ALSO
       cyaml_delete (macro), _cyaml_get, _cyaml_set_typed

NAME
       create_cthread_pool_mp - create a thread pool with custom memory
                                management

SYNOPSIS
       #include <cthreadpool.h>

       ctpool create_cthread_pool_mp(size_t num_threads,
                                     size_t queue_capacity,
                                     ccol_memmgmt_procs_t *mprocs,
                                     char **err_str);

DESCRIPTION
       Creates a new thread pool with the specified number of worker threads.
       The queue_capacity parameter controls the task queue mode:

         0 or ccol_invalid_size -> unbounded task queue. ctpool_submit never
           blocks on capacity; the only non-argument failure path is
           ccol_not_enough_memory.

         N > 0 -> bounded task queue of capacity N. ctpool_submit blocks when
           the queue is full; ctpool_try_submit returns ccol_container_full
           instead of blocking.

       All num_threads worker threads are started before the function returns.
       If any pthread_create call fails the already-started threads are joined
       cleanly before the function returns NULL.

PARAMETERS
       num_threads    Number of worker threads (must be >= 1)
       queue_capacity 0 or ccol_invalid_size for unbounded; N > 0 for bounded
       mprocs         Custom allocator, or NULL for malloc/free
       err_str        Optional: receives a static error string on failure

RETURN VALUE
       Returns a new ctpool handle on success.
       Returns NULL on failure and sets *err_str if err_str is non-NULL.

NOTES
       Futures always use standard malloc/free regardless of mprocs.
       The pool must be shut down with ctpool_shutdown_drain or
       ctpool_shutdown_immediate before calling ctpool_destroy, unless
       ctpool_destroy is used directly (which performs a drain shutdown).

SEE ALSO
       create_cthread_pool, ctpool_construct, __ctpool_destroy,
       ctpool_shutdown_drain, ctpool_shutdown_immediate

NAME

DESCRIPTION
       Allocates a task struct and appends it to the pool's task queue. For
       unbounded queues the call returns as soon as the task is enqueued and
       never blocks on capacity. For bounded queues the call blocks until a
       slot is available in the queue.

       The worker thread that picks up the task calls fn(arg). If on_complete
       is non-NULL the worker calls on_complete(arg) immediately after fn
       returns. The pool never frees arg; ownership remains with the caller.

PARAMETERS
       pool         Thread pool handle
       fn           Task function (must not be NULL)
       arg          Passed verbatim to fn and on_complete (may be NULL)
       on_complete  Called by the worker after fn returns (may be NULL)

RETURN VALUE
       ccol_success           - task enqueued successfully.
       ccol_invalid_args      - pool or fn is NULL.
       ccol_not_enough_memory - task struct allocation failed.
       ccol_not_permitted     - pool is shutting down.

SEE ALSO
       ctpool_try_submit, ctpool_timed_submit, ctpool_submit_future,
       ctpool_wait

NAME

------------------------------------------------------------------------------
NAME
       ctpool_submit_future - submit a task that returns a void* result
                              (blocking)

SYNOPSIS
       #include <cthreadpool.h>

       ctpool_future *ctpool_submit_future(ctpool pool,
                                           void *(*fn)(void *),
                                           void *arg);

DESCRIPTION
       Allocates a future object and submits a task to the pool. The task
       calls fn(arg) on a worker thread and stores the returned void* pointer
       in the future for later retrieval by ctpool_future_get.

       The returned future has a reference count of 2 on success: one
       reference held by the caller, one held internally by the queued task.
       The caller must call ctpool_future_free exactly once when done.

       For bounded queues this function blocks until a slot is available.
       Use ctpool_try_submit_future or ctpool_timed_submit_future to avoid
       blocking.

       Futures always use standard malloc/free regardless of any custom
       allocator configured at pool creation.

PARAMETERS
       pool  Thread pool handle
       fn    Task function returning void* (must not be NULL)
       arg   Passed to fn (may be NULL)

RETURN VALUE
       Returns a new ctpool_future handle on success.
       Returns NULL if pool or fn is NULL, if the pool is shutting down, or
       if memory allocation fails.

SEE ALSO
       ctpool_try_submit_future, ctpool_timed_submit_future,
       ctpool_future_get, ctpool_future_done, ctpool_future_cancelled,
       ctpool_future_free

NAME
       ctpool_future_free - release the caller's reference to a future

SYNOPSIS
       #include <cthreadpool.h>

       void ctpool_future_free(ctpool_future *f);

DESCRIPTION
       Decrements the future's reference count. When the count reaches zero
       (both the caller and the worker have released their references) the
       future object is freed with standard free().

       Must be called exactly once per successful ctpool_submit_future,
       ctpool_try_submit_future, or ctpool_timed_submit_future call. May be
       called before or after ctpool_future_get. Calling before
       ctpool_future_get is safe: the task still runs and the future is freed
       automatically when the worker finishes.

       Calling ctpool_future_free on a NULL pointer is a no-op.

       f   Future handle

       ctpool_timed_submit_future, ctpool_future_get

NAME
       ctpool_wait - block until the pool is idle

SYNOPSIS
       #include <cthreadpool.h>

       void ctpool_wait(ctpool pool);

DESCRIPTION
       Blocks until the task queue is empty and all active tasks have
       finished executing. Does not shut down the pool; worker threads
       remain alive and new tasks may be submitted after this call returns.

       Contract: no other thread should be submitting tasks concurrently
       when ctpool_wait is called. With an unbounded queue, concurrent
       submissions can cause indefinite blocking.

PARAMETERS
       pool   Thread pool handle (NULL is a no-op)

SEE ALSO
       ctpool_shutdown_drain, ctpool_shutdown_immediate

------------------------------------------------------------------------------
NAME
       ctpool_shutdown_drain - graceful shutdown

SYNOPSIS
       #include <cthreadpool.h>

       void ctpool_shutdown_drain(ctpool pool);

DESCRIPTION
       Signals the pool to stop accepting new tasks, waits for all queued
       and active tasks to complete, then joins all worker threads. Blocks
       until every worker has exited.

       After this call returns, no new tasks may be submitted. The pool
       struct can then be freed with ctpool_destroy. Calling
       ctpool_shutdown_drain a second time is a no-op.

PARAMETERS
       pool   Thread pool handle (NULL is a no-op)

NOTES
       ctpool_submit and ctpool_submit_future return ccol_not_permitted after
       this call.
       Any thread blocked in ctpool_submit on a full bounded queue is woken
       and will receive ccol_not_permitted.

SEE ALSO
       ctpool_shutdown_immediate, ctpool_wait, ctpool_destroy

------------------------------------------------------------------------------
NAME
       ctpool_shutdown_immediate - immediate shutdown

SYNOPSIS
       #include <cthreadpool.h>

       void ctpool_shutdown_immediate(ctpool pool);

DESCRIPTION
       Cancels all tasks currently in the queue, then signals worker threads
       to exit after their current task (if any) completes. Blocks until all
       worker threads have exited.

       Futures for cancelled tasks are marked done and cancelled; any thread
       blocked in ctpool_future_get for a cancelled future is unblocked and
       receives NULL. Calling ctpool_shutdown_immediate a second time is a
       no-op.

PARAMETERS
       pool   Thread pool handle (NULL is a no-op)

NOTES
       Tasks that are already executing when this is called are allowed to
       finish normally.
       ctpool_submit and ctpool_submit_future return ccol_not_permitted after
       this call.

SEE ALSO
       ctpool_shutdown_drain, ctpool_future_cancelled, ctpool_destroy

NAME
       chttpclient_do - perform an HTTP request and buffer the full response

SYNOPSIS
       #include <chttpclient.h>

       ccol_retval_t chttpclient_do(chttpcli cli,
                                    const chttp_request_t *req,
                                    chttpcli_response **resp_out);

DESCRIPTION
       Acquires a pool slot (blocking if all slots are in use), executes the
       HTTP request synchronously, buffers the entire response body, and
       stores a heap-allocated chttpcli_response in *resp_out.

       The caller owns *resp_out on success and must call
       chttpclient_resp_free when done.

       Redirects (301, 302, 303, 307, 308) are followed automatically, up to
       50 hops.  301/302/303 rewrite the method to a bodyless GET (HEAD is
       left as HEAD); 307/308 preserve the original method and body.

PARAMETERS
       cli       Client handle; must not be NULL.
       req       Request to execute; must not be NULL.
       resp_out  Output parameter; must not be NULL.

RETURN VALUE
       ccol_success
              Request completed; *resp_out is valid.

       ccol_invalid_args
              Any argument is NULL.

       ccol_not_enough_memory
              Memory allocation failed.

       ccol_timed_out
              Request or connect timeout triggered.

       ccol_not_permitted
              Client is being destroyed; no new requests are accepted.

       ccol_http_invalid_url
              URL is malformed, uses an unsupported scheme (only http:// and
              https:// are supported), or has a missing/invalid host or port.

       ccol_http_host_resolution_failed
              DNS resolution failed for the target host.

       ccol_http_connection_failed
              The TCP connection could not be established (e.g. connection
              refused).

       ccol_http_too_many_redirects
              The redirect chain exceeded 50 hops.

       ccol_http_tls_handshake_failed
              The TLS handshake failed for a reason other than certificate
              verification.

       ccol_http_tls_cert_verification_failed
              The peer certificate or hostname could not be verified, or the
              configured client certificate/key/CA bundle path was not
              readable.

       ccol_http_transfer_aborted
              The connection failed mid-transfer, the server sent a
              malformed HTTP/1.1 response, or a streaming chttpcli_write_fn
              returned fewer bytes than it was given.

       ccol_unexpected_failure
              Any other internal failure not covered by the above codes.

SEE ALSO
       chttpclient_do_streaming, chttp_do, chttpclient_resp_free

NAME
       chttpclient_do_streaming - perform an HTTP request with a streaming

NAME
       __chttpsvr_destroy - free an HTTP server

SYNOPSIS
       #include <chttpserver.h>

       void __chttpsvr_destroy(chttpsvr srv);

DESCRIPTION
       Drains all in-flight requests for this server (up to 30 seconds), then
       frees all routers, routes, middleware nodes, the server's owned ctpool,
       and the server struct itself.  If this is the last active server, also
       stops the shared facil.io engine and joins the engine thread before
       freeing.

       derived copy of it.

       first to close their listeners, then call __chttpsvr_destroy on each
       handle in any order.

       macro for that.

       Safe to call with NULL.

NAME
       chttpsvr_set_engine_mem_mgmt_procs - set custom memory management for
       engine-level allocations

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_set_engine_mem_mgmt_procs(ccol_memmgmt_procs_t *mp);

DESCRIPTION
       Redirects every allocation made by the shared facil.io engine itself
       (the connection-state table, protocol/listener structs, TLS connection
       objects, request-body streaming buffers, and the engine's own internal
       memory arena) to the malloc/free/calloc/realloc supplied in mp, exactly
       like the memory management procs accepted by every other module in this
       library.  Passing NULL reverts to the default behavior (facio's own
       arena, falling back to libc for a handful of auxiliary allocations).

       Unlike chttpsvr_set_engine_logger, this function may only be called
       before the first chttpsvr_start call in the process: once the engine
       has allocated memory with one set of procs, swapping to a different
       malloc/free pairing would corrupt the heap.  It may be called again
       after the engine has fully stopped (chttpsvr_engine_wait has returned),
       before the next chttpsvr_start.

       If mp is non-NULL, all four function pointers (malloc, free, calloc,
       realloc) must be set.

       A small, fixed amount of memory is always allocated once via mmap at
       library load time, before any call to this function is possible; this
       is harmless and freed normally at process exit, but it means a custom
       allocator can never observe every single byte ever mapped by the
       process.

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if mp is non-NULL but has a NULL function pointer.
       ccol_not_permitted if the engine is already running; stop it first.

       chttpsvr_set_engine_logger, chttpsvr_start, chttpsvr_engine_stop,
       chttpsvr_engine_wait

NAME
       chttpsvr_start - register a listener with the running engine

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_start(chttpsvr srv,
                                    const chttpsvr_config_t *cfg);

DESCRIPTION
       Starts the server on the port and host specified in cfg.  cfg may be
       NULL, in which case the defaults from CHTTPSVR_CONFIG_DEFAULT are used.

       If the shared facil.io engine is not yet running it is started
       implicitly on this call.  In this case the listener socket is bound
       and registered BEFORE the engine thread is spawned, so that the port
       is accepting connections by the time the reactor enters its event loop.
       chttpsvr_start blocks until the engine has signalled readiness (the
       FIO_CALL_ON_START callbacks have fired and the reactor is running).
       The function does NOT return until the server is fully ready to accept
       connections.

       For subsequent chttpsvr_start calls (engine already running), the
       listener socket is bound synchronously on the calling thread via
       http_listen and the function returns as soon as the socket is in
       LISTEN state.  No additional wait is needed in either case.

       The engine is kept running until the last server is destroyed.

       Multiple chttpsvr handles may be started on different ports
       simultaneously, all sharing the same engine.

       Port 0 is rejected with ccol_invalid_args.  The OS-assigned ephemeral
       port that would result from binding port 0 cannot be discovered through
       this API, making the server unreachable by clients.

       A ctpool for dispatching all handlers (buffered and streaming) is
       created at start time using cfg->worker_thread_count threads and a
       queue whose capacity is cfg->worker_queue_capacity (0 = 1024 * threads;
       CHTTPSVR_QUEUE_UNBOUNDED = no capacity limit).  The pool is owned by
       the server and destroyed in __chttpsvr_destroy.

       Body size limit: cfg->max_body_size caps the request body.  A buffered
       route whose body exceeds this is rejected with 413 before the handler
       ever runs; a streaming route's handler is always invoked, and
       chttpsvr_req_read() returns -1 with chttpsvr_req_stream_error() ==
       ccol_msg_too_large once the limit is crossed.  Either way the
       connection is closed after the resulting response (Connection: close)
       rather than kept alive, since the excess body bytes beyond the limit
       are discarded, not drained.

       Body read timeouts: cfg->stream_read_timeout_ms bounds how long a
       worker will wait for the next batch of body bytes (buffered or
       streaming); 0 waits indefinitely.  Because it resets on any new byte,
       a client that trickles data just fast enough never trips it, so
       cfg->max_body_read_duration_ms (0 = disabled, the default) separately
       bounds the *total* time spent reading one request's body regardless of
       per-gap progress -- see chttpsvr_req_read.

       Connection timeout: both cfg->read_timeout_ms and cfg->idle_timeout_ms
       map to the single connection-timeout parameter exposed by the underlying
       facil.io http_listen call.  When idle_timeout_ms is non-zero it takes
       precedence over read_timeout_ms; otherwise read_timeout_ms is used.
       Both values are converted to whole seconds (rounded up from 1 ms,
       clamped to 255 s).  A value of 0 for both fields uses the facil.io
       default of approximately 40 seconds.

       Restart: after chttpsvr_stop is called, chttpsvr_start may be called
       again on the same handle to register a new listener.  The old ctpool is
       drained and replaced with a fresh one.  Any TLS context from the
       previous cycle is freed automatically before the new one is installed.

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if srv is NULL or cfg->port is 0.
       ccol_not_permitted if this server is already started.
       ccol_not_enough_memory if memory allocation fails.
       ccol_unexpected_failure on http_listen errors.

SEE ALSO
       chttpsvr_stop, chttpsvr_engine_stop, chttpsvr_set_engine_logger,
       chttpsvr_set_engine_mem_mgmt_procs, chttpsvr_engine_wait,
       __chttpsvr_destroy

NAME
       chttpsvr_register_handler - register a buffered handler on the root router

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_register_handler(chttpsvr srv, chttp_method_t method,
                                  const char *pattern,
                                  chttpsvr_handler_fn fn, void *ctx);

DESCRIPTION
       Registers fn as the handler for requests matching method and pattern
       on the root router.  pattern uses forward-slash-separated segments;
       segments of the form {name} are named path parameters.

       Passing CHTTP_ANY for method registers a wildcard route that matches
       every HTTP method on the given pattern.  The handler can call
       chttpsvr_req_method to retrieve the actual incoming method.
       Registration order determines priority: a method-specific route
       registered before a CHTTP_ANY route on the same pattern wins for its
       method; the CHTTP_ANY route catches every other method.

       pattern must begin with '/'.  Patterns that do not start with '/' are
       rejected immediately with ccol_invalid_args.

       The name inside a {name} segment must consist entirely of characters
       from [A-Za-z0-9_].  Names containing other characters (spaces, hyphens,
       dots, etc.) are rejected with ccol_invalid_args because they can never
       be retrieved via chttpsvr_req_param.

       Path-segment values (both literal segments and captured parameters) are
       decoded using RFC 3986 path semantics: percent-encoded sequences such as
       %20 are decoded, but '+' is left as a literal '+'.  This is consistent
       with the decoded path returned by chttpsvr_req_path.  It differs from
       query-string (application/x-www-form-urlencoded) decoding, where '+' is
       treated as a space; that convention applies only to chttpsvr_req_query
       and chttpsvr_req_query_one, not to path parameters.

       The router scans all registered routes looking for a path-and-method
       match.  Multiple methods may be registered for the same path; all are
       reachable regardless of registration order.  If at least one route
       matches the path but none matches the method, the server responds 405.
       If no route matches the path, the server responds 404.  Path matching
       includes validation of percent-encoded sequences: a request with invalid
       encoding in any path segment -- whether a literal segment (e.g.
       /bad%ZZusers/{id} against /users/{id}) or a captured {name} parameter
       (e.g. /users/bad%ZZvalue against /users/{id}) -- does not match the
       route and returns 404 regardless of which methods are registered for
       that pattern.

       If the same method and pattern combination is registered more than once,
       each call succeeds (returns ccol_success) but only the FIRST registered
       handler is ever invoked (first-wins policy).

       Trailing slashes are NOT treated as equivalent: a request for
       /users/42/ does not match the pattern /users/{id}.  Register a
       separate pattern if the trailing-slash form must be accepted.

       Patterns with consecutive slashes (e.g. /foo//bar) or a trailing slash
       (e.g. /foo/) are rejected at registration time.  A trailing slash in an
       incoming request path does NOT match a registered pattern that lacks one;
       register a separate pattern if the trailing-slash form must be accepted.

       This function is thread-safe and may be called before or after
       chttpsvr_start is running.

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if srv, pattern, or fn is NULL; if pattern does not
                         begin with '/'; if pattern contains consecutive
                         slashes or a trailing slash; or if a {name} segment
                         contains characters outside [A-Za-z0-9_].
       ccol_not_enough_memory if allocation fails.

SEE ALSO
       chttpsvr_register_streaming_handler, chttpsvr_use, chttpsvr_subrouter

------------------------------------------------------------------------------
NAME
       chttpsvr_register_streaming_handler - register a streaming handler on the root router

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_register_streaming_handler(chttpsvr srv,
                                         chttp_method_t method,
                                         const char *pattern,
                                         chttpsvr_handler_fn fn, void *ctx);

DESCRIPTION
       Like chttpsvr_register_handler but marks the route as streaming, which
       makes chttpsvr_req_read available to the handler for sequential body
       consumption.  The handler runs on the server's owned ctpool worker
       thread (not on the facil.io reactor thread) -- exactly as all handlers
       do in this implementation -- but chttpsvr_req_read returns -1 on
       non-streaming routes, so the streaming flag controls the API contract.

       CHTTP_ANY may be used for method to match every HTTP method on the
       pattern.  See chttpsvr_register_handler for the full CHTTP_ANY
       semantics.

       Routing happens as soon as headers are parsed, before any body byte is
       read, so the worker (never the reactor thread) reads the body itself,
       batch by batch, directly off the socket. chttpsvr_req_read hands each
       batch to the handler as it arrives and returns 0 once the body ends;
       subsequent calls also return 0 (idempotent EOF). It returns -1 on a
       dropped connection, an exceeded max_body_size, or a stalled client
       (bounded by chttpsvr_config_t.stream_read_timeout_ms per gap and,
       if set, chttpsvr_config_t.max_body_read_duration_ms overall) -- call
       chttpsvr_req_stream_error to distinguish these.

       When the server's ctpool queue is at capacity when a request arrives,
       the server responds with 503 Service Unavailable immediately rather
       than blocking the reactor thread.

       The same pattern rules apply as for chttpsvr_register_handler: pattern
       must begin with '/'; patterns with consecutive slashes or a trailing
       slash are rejected; and {name} segments must contain only [A-Za-z0-9_].
       See chttpsvr_register_handler for full details on pattern validation and
       path-segment decoding semantics.

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if srv, pattern, or fn is NULL; if pattern does not
                         begin with '/'; if pattern contains consecutive
                         slashes or a trailing slash; or if a {name} segment
                         contains characters outside [A-Za-z0-9_].
       ccol_not_enough_memory if allocation fails.

SEE ALSO
       chttpsvr_register_handler, chttpsvr_req_read

NAME
       chttpsvr_use - append global middleware to the root router

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_use(chttpsvr srv,
                                   chttpsvr_middleware_fn fn, void *ctx);

DESCRIPTION
       Appends a middleware function to the global middleware chain.  Global
       middleware runs before router-specific middleware and before the route
       handler for every request.

       fn receives a next callback; calling it continues the chain.  Omitting
       the call short-circuits the chain and sends the response as-is.

       This function is thread-safe and may be called before or after
       chttpsvr_start is running.

NOTES
       Each router (the root router for global middleware, and each
       sub-router) caps its own middleware chain at 32 entries.  This cap is
       enforced at REGISTRATION time: chttpsvr_use and chttpsvr_router_use
       return ccol_not_permitted, without adding the entry, on the call that
       would exceed the 32-entry cap for that specific router.

       Separately, dispatch time also enforces a combined cap of 32 for the
       effective chain of a given request (global middleware + the matched
       router's own middleware).  Because each side of that sum can
       independently hold up to 32 entries, the combined count can still
       exceed 32 even though neither router individually hit its own
       registration-time cap; when this happens the server responds with
       HTTP 500 on the affected request.

       Registering the same path and method more than once is permitted and
       returns ccol_success each time, but only the FIRST registered handler
       is ever invoked (first-wins policy).

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if srv or fn is NULL.
       ccol_not_enough_memory if allocation fails.
       ccol_not_permitted if the root router's middleware chain is already at
       the 32-entry cap.

SEE ALSO
       chttpsvr_register_handler, chttpsvr_router_use

NAME
       chttpsvr_subrouter - create a sub-router with a path prefix

SYNOPSIS
       #include <chttpserver.h>

       chttpsvr_router *chttpsvr_subrouter(chttpsvr srv, const char *prefix);

DESCRIPTION
       Creates a new router under the given path prefix (which must begin with
       '/') and attaches it to the server.  Routes registered on the sub-router
       are matched against the portion of the path following the prefix.

       The prefix must not contain consecutive slashes (e.g. "//api" or
       "/a//b").  Such prefixes can never match a valid HTTP path and would
       silently create an unreachable router; chttpsvr_subrouter rejects them
       by returning NULL.

       A trailing slash in prefix is stripped automatically: "/api/v1/" and
       "/api/v1" are therefore equivalent and produce identical routing
       behaviour.

       The returned pointer is owned by the server and must not be freed by
       the caller.

       This function is thread-safe and may be called before or after
       chttpsvr_start is running.

NOTES
       The "/" prefix is a special case.  The trailing-slash strip loop runs
       only when the prefix length exceeds one character, so "/" is retained
       as-is with prefix_len = 1.  The match rule checks path[prefix_len],
       i.e. path[1], which must be '/' or '\0'.  For the exact path "/" the
       character at index 1 is '\0' (matches).  For any other path beginning
       with '/' (e.g. "/foo"), path[1] is a letter, which fails the check:
       those paths do NOT match and receive 404.

       Use prefix "/" only when you intend to match only the exact root path
       "/".  To catch all requests regardless of path, register routes directly
       on the server with chttpsvr_register_handler or chttpsvr_register_streaming_handler instead.

       Root-router shadowing: routes registered directly on the server via
       chttpsvr_register_handler or chttpsvr_register_streaming_handler (i.e. on the root router) are always
       evaluated before any sub-router.  A root-level route whose path
       conflicts with a sub-router pattern will always win, regardless of
       registration order.  To avoid unintentional shadowing, do not register
       root-level routes whose paths overlap with a sub-router's prefix and
       pattern.

RETURN VALUE
       Pointer to the new chttpsvr_router on success, or NULL when srv is NULL,
       prefix is NULL, prefix does not start with '/', prefix contains
       consecutive slashes, or on OOM.

SEE ALSO
       chttpsvr_router_on, chttpsvr_router_use

NAME
       chttpsvr_req_read - read body bytes as they arrive (streaming handlers)

SYNOPSIS
       #include <chttpserver.h>

       ssize_t chttpsvr_req_read(chttpsvr_req *req, void *buf, size_t buflen);

DESCRIPTION
       Reads up to buflen bytes of the request body into buf, batch by batch,
       as they actually arrive on the connection.  This call reads the socket
       itself -- on the worker thread executing the handler, never the
       reactor thread -- and blocks the calling worker until at least one
       byte is available, the body ends, an error occurs,
       chttpsvr_config_t.stream_read_timeout_ms elapses without new data, or
       (if set) chttpsvr_config_t.max_body_read_duration_ms elapses.

       Return values:
         > 0   Number of bytes copied into buf.
           0   End of body reached, or buflen is 0 (valid no-op).
          -1   req is NULL; buf is NULL with buflen > 0; the handler was
               registered with chttpsvr_register_handler (buffered), not
               chttpsvr_register_streaming_handler; the connection was
               dropped or reset; stream_read_timeout_ms or
               max_body_read_duration_ms elapsed; or the body exceeded
               max_body_size mid-stream.  Call chttpsvr_req_stream_error
               immediately afterward to distinguish these.

       When buflen is 0 the function returns 0 immediately regardless of the
       value of buf.  Passing NULL for buf with buflen == 0 is therefore
       valid and returns 0 (a no-op), consistent with POSIX read(2)
       semantics.  Only when buflen > 0 is buf required to be non-NULL.

       This function is intended for streaming handlers registered with
       chttpsvr_register_streaming_handler.  For buffered handlers, use chttpsvr_req_body.

SEE ALSO
       chttpsvr_req_body, chttpsvr_req_stream_error, chttpsvr_register_streaming_handler

NAME
       chttpsvr_req_stream_error - report why the last chttpsvr_req_read call failed

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_req_stream_error(const chttpsvr_req *req);

DESCRIPTION
       Reports the reason the most recent chttpsvr_req_read call on req
       returned -1.  Meaningful only immediately after such a -1 return;
       otherwise the result is unspecified.

       Return values:
         ccol_timed_out             stream_read_timeout_ms elapsed while
                                     waiting for the next batch, or the
                                     request's total body read time exceeded
                                     max_body_read_duration_ms (if set).
         ccol_msg_too_large         The body exceeded max_body_size
                                     mid-stream.
         ccol_http_transfer_aborted The connection was closed or reset, or
                                     the body's framing was malformed.
         ccol_success               No error is on record for req.
         ccol_unexpected_failure    req is NULL or its handle is invalid.

SEE ALSO

NAME
       chttpsvr_req_query - look up all values for a query parameter

SYNOPSIS
       #include <chttpserver.h>

       const char **chttpsvr_req_query(chttpsvr_req *req, const char *key,
                                        size_t *count_out);

DESCRIPTION
       Parses the query string lazily on first call and returns a
       NULL-terminated array of all URL-decoded values associated with key.
       If count_out is non-NULL, stores the count.  Returns NULL if the key
       is not present.

       The string values pointed to by the array elements are valid for the
       lifetime of the handler invocation.  The array pointer itself is a
       per-request scratch buffer that is recycled on every call to
       chttpsvr_req_query: copy any pointers you need from the returned array
       before making a subsequent call to chttpsvr_req_query on the same
       request.

NOTES
       This function cannot distinguish an OOM failure (during query-string
       parsing or while growing the result array) from a genuine key-absent
       result.  Both conditions produce (NULL, count=0).  Call
       chttpsvr_req_query_oom(req) after a NULL return to distinguish the two
       cases.  Alternatively, use chttpsvr_req_query_one for single-valued
       keys; that function returns ccol_not_enough_memory on OOM.

SEE ALSO
       chttpsvr_req_query_one, chttpsvr_req_query_oom, chttpsvr_req_raw_query

NAME
       chttpsvr_req_query_oom - check whether chttpsvr_req_query ran out of memory

SYNOPSIS
       #include <chttpserver.h>

       bool chttpsvr_req_query_oom(const chttpsvr_req *req);

DESCRIPTION
       Returns true if chttpsvr_req_query could not grow its internal result
       array due to an allocation failure during the current request.

       chttpsvr_req_query returns (NULL, count=0) both when the requested key
       is absent and when an OOM condition prevents building the result array.
       Use this function to distinguish the two cases after any NULL return

NAME
       chttpsvr_resp_set_header - set a response header

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_resp_set_header(chttpsvr_resp *resp,
                                               const char *name,
                                               const char *value);

DESCRIPTION
       Sets or replaces a response header.  If a header with the same name
       already exists (case-insensitive comparison per RFC 7230), the stored
       value is replaced in-place and the function returns immediately.

       Name casing: when a duplicate is detected the stored header name retains
       the casing supplied by the FIRST call; the casing of subsequent calls is
       ignored.  HTTP/1.1 header names are case-insensitive on the wire, so
       this does not affect correctness, but callers who read back stored header
       names directly should account for this behaviour.

       All three arguments must be non-NULL; passing NULL for any of them
       returns ccol_invalid_args without modifying the response.

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if resp, name, or value is NULL.
       ccol_not_enough_memory if the underlying allocation fails.

       chttpsvr_resp_set_status, chttpsvr_resp_write

NAME
       chttpsvr_resp_write - append bytes to the response body

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_resp_write(chttpsvr_resp *resp,
                                          const void *data, size_t len);

DESCRIPTION
       Appends len bytes of data to the internal response body buffer.  The
       buffer grows automatically using an overflow-safe doubling strategy:
       capacity is at least doubled on each reallocation and the computation
       is guarded against size_t wrap-around.  ccol_not_enough_memory is
       returned both when len would overflow size_t when added to the current
       body length and when the underlying allocator returns NULL.

RETURN VALUE
       ccol_success on success.
       ccol_invalid_args if resp is NULL, or if data is NULL and len > 0.
       ccol_not_enough_memory if the body length would overflow size_t or if
                              the allocator cannot satisfy the request.

SEE ALSO
       chttpsvr_resp_write_str, chttpsvr_resp_write_json

NAME
       chttpsvr_resp_write_json - append JSON to the response body

SYNOPSIS
       #include <chttpserver.h>

       ccol_retval_t chttpsvr_resp_write_json(chttpsvr_resp *resp,
                                               const char *json, size_t len);

DESCRIPTION
       Appends len bytes of json to the response body and then sets the
       Content-Type response header to "application/json".  The body is
       written first so that an OOM failure on body allocation leaves resp
       completely unchanged (no header is installed, no bytes are buffered).

       Returns ccol_invalid_args immediately (without modifying resp) when
       resp or json is NULL, or when len is 0.  A zero-length JSON body with
       Content-Type set is semantically invalid, so the function rejects it
       rather than silently setting the header with no body bytes.

RETURN VALUE
       ccol_success           - Body bytes appended and Content-Type header set.
       ccol_invalid_args      - resp or json is NULL, or len is 0.
       ccol_not_enough_memory - Body or header buffer allocation failed.

       chttpsvr_resp_write, chttpsvr_resp_set_header
