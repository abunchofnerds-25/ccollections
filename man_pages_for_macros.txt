------------------------------------------------------------------------------
NAME
       cvector_destroy - destroy a vector and set pointer to NULL

SYNOPSIS
       #include <cvector.h>

       cvector_destroy(v)

DESCRIPTION
       Frees all resources associated with the vector including the data buffer.
       The vector pointer is automatically set to NULL after destruction. This
       is a macro wrapper around __cvector_destroy().

PARAMETERS
       v    Vector to destroy (will be set to NULL after destruction)

NOTES
       Safe to call with NULL pointer (no-op)
       Does not free individual elements when the vector contains pointers
       to heap-allocated buffers - caller must free element data first if needed

SEE ALSO
       cvector_create, __cvector_destroy

------------------------------------------------------------------------------
NAME
       cvec_construct - type-safe vector construction

SYNOPSIS
       #include <cvector.h>

       cvec_construct(vec_name, type);

DESCRIPTION
       Type-safe macro for creating a vector. Automatically determines element
       size from the type and creates type marker variables for compile-time
       type checking in other macros.

PARAMETERS
       vec_name  Name of the vector variable to create
       type      Element type (e.g., int, double, struct foo)

NOTES
       Terminates program on allocation failure
       Creates hidden type marker variable for type safety
       Equivalent to cvector_create(sizeof(type), &err) with error handling

EXAMPLES
       cvec_construct(my_vec, int);
       cvec_push_rvalue(my_vec, 42);

SEE ALSO
       cvec_construct_scoped, cvector_create

------------------------------------------------------------------------------
NAME
       cvec_push - push an element onto the vector (type-safe)

SYNOPSIS
       #include <cvector.h>

       cvec_push(v, new_elem);

DESCRIPTION
       Type-safe wrapper for cvector_push_back() that automatically takes the
       address of the element. Calls fatal_err() on failure.

PARAMETERS
       v         Vector to push onto
       new_elem  Element to push (passed by reference automatically)

NOTES
       Terminates program on failure
       Automatically takes address of new_elem
       Type is verified at compile time

EXAMPLES
       cvec_construct(vec, int);
       int x = 42;
       cvec_push(vec, x);

SEE ALSO
       cvec_push_rvalue, cvector_push_back

------------------------------------------------------------------------------
NAME
       cvec_push_rvalue - push an rvalue onto the vector (type-safe)

SYNOPSIS
       #include <cvector.h>

       cvec_push_rvalue(v, new_elem);

DESCRIPTION
       Type-safe wrapper for cvector_push_back() that accepts rvalues (literals,
       expressions). Creates a temporary variable and takes its address. Calls
       fatal_err() on failure.

PARAMETERS
       v         Vector to push onto
       new_elem  Expression or literal to push

NOTES
       Terminates program on failure
       Accepts expressions and literals
       Uses GNU C statement expression extension

EXAMPLES
       cvec_construct(vec, int);
       cvec_push_rvalue(vec, 42);
       cvec_push_rvalue(vec, x + y);

SEE ALSO
       cvec_push, cvector_push_back

------------------------------------------------------------------------------
NAME
       cvec_pop - pop an element from the vector (type-safe)

SYNOPSIS
       #include <cvector.h>

       value = cvec_pop(v);

DESCRIPTION
       Type-safe wrapper for cvector_pop_back() that returns the popped element
       as a value. Calls fatal_err() on failure.

PARAMETERS
       v    Vector to pop from

RETURN VALUE
       Returns the popped element value

NOTES
       Terminates program on failure
       Returns value, not pointer
       Uses GNU C statement expression extension

EXAMPLES
       cvec_construct(vec, int);
       cvec_push_rvalue(vec, 42);
       int val = cvec_pop(vec);  // val == 42

SEE ALSO
       cvector_pop_back, cvec_push

------------------------------------------------------------------------------
NAME
       cvec_at - access element at index (type-safe, returns reference)

SYNOPSIS
       #include <cvector.h>

       element_ref = cvec_at(v, index);

DESCRIPTION
       Type-safe wrapper for cvector_at() that returns the element reference
       (not pointer). Automatically casts to the correct type. Can be used as
       lvalue for assignment.

PARAMETERS
       v      Vector to access
       index  Zero-based index of element

RETURN VALUE
       Returns element reference at index

NOTES
       Returns reference, not pointer
       No bounds checking - will cause SIGSEGV if index out of bounds
       Can be used on left side of assignment

EXAMPLES
       cvec_construct(vec, int);
       cvec_push_rvalue(vec, 42);
       int val = cvec_at(vec, 0);  // val == 42
       cvec_at(vec, 0) = 5;        // now the first element is 5

SEE ALSO
       cvector_at

------------------------------------------------------------------------------
NAME
       cvector_sort_with_comparison_proc - sort vector with custom comparison

SYNOPSIS
       #include <cvector.h>

       cvector_sort_with_comparison_proc(v, comparison_proc);

DESCRIPTION
       Sorts the vector in-place using the csort library with a custom comparison
       function. The comparison function should follow the standard comparator
       convention (return <0, 0, or >0).

PARAMETERS
       v                 Vector to sort
       comparison_proc   Comparison function for sorting

NOTES
       Sorts in-place using stable mergesort algorithm
       O(n log n) time complexity, O(n) space complexity
       Asserts if v is NULL
       Uses csort for sorting

EXAMPLES
       int compare_ints(const void *a, const void *b) {
         return *(int*)a - *(int*)b;
       }
       cvec_construct(vec, int);
       // ... add elements ...
       cvector_sort_with_comparison_proc(vec, compare_ints);

SEE ALSO
       cvec_sort, csort_sort

------------------------------------------------------------------------------
NAME
       cvec_sort - sort vector using default comparison for type

SYNOPSIS
       #include <cvector.h>

       cvec_sort(v);

DESCRIPTION
       Sorts the vector in-place using the default comparison function for the
       element type. The default comparator is obtained from the csort library
       based on the type.

PARAMETERS
       v    Vector to sort

NOTES
       Sorts in-place using stable mergesort algorithm
       O(n log n) time complexity, O(n) space complexity
       Uses default comparison for the element type
       Supported types depend on csort library defaults

EXAMPLES
       cvec_construct(vec, int);
       cvec_push_rvalue(vec, 3);
       cvec_push_rvalue(vec, 1);
       cvec_push_rvalue(vec, 2);
       cvec_sort(vec);  // vec is now [1, 2, 3]

SEE ALSO
       cvector_sort_with_comparison_proc

                        CSORT.H - SORTING LIBRARY
================================================================================

------------------------------------------------------------------------------
NAME
       csort_sort - sort any collection with custom getter and comparison

SYNOPSIS
       #include <csort.h>

       csort_sort(collection, length, elem_size, getter_proc, 
                  comparison_proc, mprocs);

DESCRIPTION
       Generic sorting macro that sorts any collection using an iterative
       mergesort algorithm. Works with any collection type (arrays, vectors,
       custom containers) as long as getter and comparison functions are provided.

       This is a wrapper macro around ___csort_merge_sort().

PARAMETERS
       collection       Pointer to collection to sort
       length          Number of elements in collection
       elem_size       Size of each element in bytes
       getter_proc     Function to get element at index
       comparison_proc Function to compare two elements
       mprocs          Memory management procedures, or NULL for default

NOTES
       Stable sort (preserves relative order of equal elements)
       O(n log n) time complexity in all cases
       O(n) space complexity for temporary merge buffer
       Iterative implementation (no recursion)

EXAMPLES
       // Sort a plain array
       int arr[] = {3, 1, 2};
       void *getter(void *col, size_t i) { return &((int*)col)[i]; }
       int cmp(const void *a, const void *b) { 
         return *(int*)a - *(int*)b; 
       }
       csort_sort(arr, 3, sizeof(int), getter, cmp, NULL);

SEE ALSO
       csort_get_default_comparison_proc, cvec_sort

------------------------------------------------------------------------------
NAME
       csort_get_default_comparison_proc - get default comparison for a type

SYNOPSIS
       #include <csort.h>

       ccol_comparison_proc_t csort_get_default_comparison_proc(dummy_var);

DESCRIPTION
       Returns the default comparison function for a given type. Uses C11 
       _Generic to select the appropriate comparison based on the type of the 
       dummy variable.

       Supports all standard C integral types (char, short, int, long, long long,
       unsigned variants) and floating-point types (float, double, long double),
       as well as strings (char*).

PARAMETERS
       dummy_var  A variable of the type you want to compare (not evaluated)

RETURN VALUE
       Returns pointer to appropriate comparison function
       Returns NULL if type is not supported

NOTES
       Variable is only used for type deduction, not evaluated
       For strings, returns csort_default_string_comparison_proc
       For integral types, uses three-way comparison: (a > b) - (a < b)

EXAMPLES
       // Get comparator for strings
       char* dummy_str;
       ccol_comparison_proc_t cmp = csort_get_default_comparison_proc(dummy_str);
       // cmp now points to csort_default_string_comparison_proc

       // Get comparator for doubles
       double dummy_double;
       cmp = csort_get_default_comparison_proc(dummy_double);

SEE ALSO
       csort_sort, cvec_sort

                         CHASHMAP.H - HASH MAP

------------------------------------------------------------------------------
NAME
       chmap_destroy - destroy a hash map and set pointer to NULL

SYNOPSIS
       #include <chashmap.h>

       chmap_destroy(hm);

DESCRIPTION
       Frees all resources associated with the hash map including all key-value
       pairs. The hash map pointer is automatically set to NULL after destruction.

PARAMETERS
       hm   Hash map to destroy (will be set to NULL after destruction)

NOTES
       Safe to call with NULL pointer
       Frees all keys and values
       For maps created from preallocated buffers, only the map structure is
       freed (user must manage buffer lifetime)

SEE ALSO
       chmap_create, __chmap_destroy

------------------------------------------------------------------------------
NAME
       chmap_construct - type-safe hash map construction

SYNOPSIS
       #include <chashmap.h>

       chmap_construct(hm_name, key_type, val_type);

DESCRIPTION
       Type-safe macro for creating a hash map with default settings. 
       Automatically determines key and value types and creates type marker
       variables for compile-time type checking.

PARAMETERS
       hm_name   Name of the hash map variable to create
       key_type  Type of keys
       val_type  Type of values

NOTES
       Terminates program on allocation failure
       Creates hidden type marker variables for type safety
       Uses DEFAULT_INITIAL_BUCKET_ARRAY_SIZE (64) buckets
       Uses default memory management and hashing

EXAMPLES
       chmap_construct(map, int, char*);
       chmap_insert(map, 42, "hello");

SEE ALSO
       chmap_construct_full, chmap_create

------------------------------------------------------------------------------
NAME
       chmap_insert - insert a key-value pair (type-safe)

SYNOPSIS
       #include <chashmap.h>

       chmap_insert(hm_name, key, val);

DESCRIPTION
       Type-safe wrapper for chmap_insert_elem() that automatically handles
       type conversion and cmap_pair creation. If the key already exists the
       value is updated silently. Calls fatal_err() only on a true failure
       (i.e. ccol_not_enough_memory or ccol_container_full).

PARAMETERS
       hm_name  Hash map to insert into
       key      Key to insert
       val      Value to associate with key

NOTES
       Terminates program only on true failure, not on existing-key update
       Automatically takes address of key and value
       Handles both value types and string types correctly

EXAMPLES
       chmap_construct(map, int, char*);
       chmap_insert(map, 42, "hello");
       chmap_insert(map, 100, "world");

SEE ALSO
       chmap_insert_elem, chmap_get, chmap_remove

------------------------------------------------------------------------------
NAME
       chmap_remove - remove a key-value pair (type-safe)

SYNOPSIS
       #include <chashmap.h>

       ccol_retval_t chmap_remove(hm_name, key);

DESCRIPTION
       Type-safe wrapper for chmap_delete_elem() that returns the result code.
       Does not terminate on key_not_found.

PARAMETERS
       hm_name  Hash map to remove from
       key      Key to remove

RETURN VALUE
       Returns ccol_success if removed
       Returns ccol_key_not_found if not found

NOTES
       Does not terminate on key_not_found
       Frees memory for key and value

EXAMPLES
       ccol_retval_t r = chmap_remove(map, 42);
       if (r == ccol_key_not_found) {
         printf("Key not found\n");
       }

SEE ALSO
       chmap_delete_elem, chmap_insert

------------------------------------------------------------------------------
NAME
       chmap_get - get value by key (type-safe, returns value)

SYNOPSIS
       #include <chashmap.h>

       value = chmap_get(hm_name, key);

DESCRIPTION
       Type-safe wrapper for chmap_get_elem_ref() that returns the actual value.
       Calls fatal_err() if key not found or size mismatch.

PARAMETERS
       hm_name  Hash map to search
       key      Key to look up

RETURN VALUE
       Returns value associated with key

NOTES
       Terminates program if key not found
       Terminates program if value size doesn't match type size
       Returns value, not pointer
       For strings, returns the char* itself

EXAMPLES
       chmap_construct(map, int, char*);
       chmap_insert(map, 42, "hello");
       char* val = chmap_get(map, 42); // val == "hello"

SEE ALSO
       chmap_get_ptr, chmap_insert

------------------------------------------------------------------------------
NAME
       chmap_get_ptr - get pointer to value by key (type-safe)

SYNOPSIS
       #include <chashmap.h>

       value_ptr = chmap_get_ptr(hm_name, key);

DESCRIPTION
       Type-safe wrapper for chmap_get_elem_ref() that returns a pointer to the
       value, or NULL if key not found. Unlike chmap_get(), does not terminate
       on key_not_found.

PARAMETERS
       hm_name  Hash map to search
       key      Key to look up

RETURN VALUE
       Returns pointer to value if key found
       Returns NULL if key not found

NOTES
       Returns NULL if key not found (does not terminate)
       Terminates program if value size doesn't match type size
       Returns pointer to value for in-place modification
       Pointer invalidated by later insert/delete/resize operations
       For strings, returns pointer to the char* itself

EXAMPLES
       chmap_construct(map, int, int);
       chmap_insert(map, 42, 100);
       int* ptr = chmap_get_ptr(map, 42);
       if (ptr) {
         *ptr = 200; // Modify in-place
       }

SEE ALSO
       chmap_get, chmap_get_elem_ref

------------------------------------------------------------------------------
NAME
       DECLARE_PREALLOCATED_MEMPOOL_BUFFER - declare buffer for memory pool

SYNOPSIS
       #include <cmempool.h>

       DECLARE_PREALLOCATED_MEMPOOL_BUFFER(name, elem_count, elem_size);

DESCRIPTION
       Declares a uint8_t array sized correctly for a memory pool with the
       specified parameters. The buffer can then be passed to
       mempool_create_from_preallocated_buffer(). Automatically includes header
       overhead in size calculation.

PARAMETERS
       name        Variable name for the buffer
       elem_count  Number of elements the pool will hold
       elem_size   Size of each element in bytes

NOTES
       Buffer is allocated on stack or in static storage depending on context
       Useful for embedded systems or avoiding heap allocation

EXAMPLES
       DECLARE_PREALLOCATED_MEMPOOL_BUFFER(my_buffer, 100, 64);
       mempool *mp = mempool_create_from_preallocated_buffer(
           my_buffer, sizeof(my_buffer), 64, false, false, NULL, NULL);

SEE ALSO
       mempool_create_from_preallocated_buffer

------------------------------------------------------------------------------
NAME
       mempool_destroy - destroy a memory pool and set pointer to NULL

SYNOPSIS
       #include <cmempool.h>

       mempool_destroy(mp);

DESCRIPTION
       Frees all resources associated with the memory pool. For pools created
       from preallocated buffers, the buffer itself is not freed (only the pool
       struct).

PARAMETERS
       mp   Memory pool to destroy (will be set to NULL after destruction)

WARNINGS
       Unfreed dynamically allocated pointers via the fallback mechanism will
       cause assertion failure. All allocated entries should be freed before
       destroying the pool.

NOTES
       Safe to call with NULL pointer
       For preallocated pools, user must manage buffer lifetime

SEE ALSO
       mempool_create, _mempool_destroy

------------------------------------------------------------------------------
NAME
       mempool_free_entry - free an entry back to the pool

SYNOPSIS

       Returns an allocated entry to the pool's free list. Macro wrapper around
       _mempool_free_entry() that sets pointer to NULL.

PARAMETERS
       entry  Entry to free (will be set to NULL after freeing)

NOTES
       O(1) complexity
       Safe to call with NULL
       Thread-safe if pool created with single_threaded=false
       Performs corruption detection via assertions

SEE ALSO
       mempool_alloc_entry, _mempool_free_entry

------------------------------------------------------------------------------
NAME
       circular_queue_create - create a circular queue

SYNOPSIS
       #include <cthreadcomm.h>

       circular_queue_create(max_size, err_str);

DESCRIPTION
       Creates a fixed-size bounded queue that uses a circular buffer internally.
       Send operations block when the queue is full until space becomes available.
       Uses standard malloc/free for memory management.

       This is a convenience macro for circular_queue_create_with_mprocs() with
       NULL memory management procedures.

PARAMETERS
       max_size  Maximum number of messages the queue can hold (1 to max_elem_count)
       err_str   Optional pointer to receive error string on failure

RETURN VALUE
       Returns pointer to newly created circular queue on success
       Returns NULL on failure

NOTES
       All operations on this queue are thread-safe
       Queue must be destroyed with circular_queue_destroy() when done

SEE ALSO
       circular_queue_create_with_mprocs, circular_queue_destroy

------------------------------------------------------------------------------

       circular_queue_destroy - destroy a circular queue

       automatically set to NULL after destruction.

PARAMETERS
       cq   Circular queue to destroy (will be set to NULL)

WARNINGS

       Drain the queue before destroying

NOTES
       Safe to call with NULL pointer

SEE ALSO

NAME
       channel_create - create a bidirectional communication channel

SYNOPSIS
       #include <cthreadcomm.h>

       channel_create(max_size, err_str);

DESCRIPTION
       Creates a bidirectional channel with two internal circular queues for
       communication between an owner thread and worker threads. Both directions
       share the same capacity. Uses standard malloc/free for memory management.

       This is a convenience macro for channel_create_with_mprocs() with NULL
       memory management procedures.

PARAMETERS
       max_size   Maximum number of messages each direction can hold
       err_str    Optional error string pointer

RETURN VALUE
       Returns pointer to newly created channel on success
       Returns NULL on failure

NOTES
       All operations are thread-safe
       Channel must be destroyed with channel_destroy() when done
       Direction is automatically selected based on calling thread ID

NAME

       #include <cthreadcomm.h>

DESCRIPTION

       wrapping the given circular_queue pointer with the specified direction.

         ccol_select_read  — wait for at least one message to arrive (read-wait)
         ccol_select_write — wait until the queue has room for a send (write-wait)

       This is a macro that expands to a compound literal; it has no run-time
       overhead beyond loading the pointer.

       For write-direction wins, ccol_select() leaves buf untouched — no message
       is consumed.  The caller must subsequently call circq_try_send_zc() since
       the available slot is not reserved (TOCTOU, same as POSIX select(2)).

            that ccol_select() will reject with ccol_invalid_args.
       dir  ccol_select_read or ccol_select_write.

RETURN VALUE
       ccol_selectable value (not a pointer).

EXAMPLES
       circular_queue *q0 = circular_queue_create(8, NULL);
       circular_queue *q1 = circular_queue_create(8, NULL);

       c_message_t msg;
       size_t idx;
       /* Read-wait on two queues: */
       ccol_select_va(&msg, &idx,
           selectable_from_circq(q0, ccol_select_read),
           selectable_from_circq(q1, ccol_select_read));

       /* Mixed: wait until q0 has room to send or q1 has a message: */
       ccol_select_va(&msg, &idx,
           selectable_from_circq(q0, ccol_select_write),
           selectable_from_circq(q1, ccol_select_read));

NAME
       selectable_from_chan - build a ccol_selectable from a channel, resolving
       the queue for the calling thread and the requested direction

SYNOPSIS
       #include <cthreadcomm.h>

       selectable_from_chan(ch, dir);

DESCRIPTION
       Calls ccol_selectable_from_chan(ch, dir), which inspects the calling
       thread's ID and the requested direction to pick the correct internal
       circular queue:

         ccol_select_read:  owner ← workers_to_owner_cq  (same as chan_recv_zc)
                          worker ← owner_to_workers_cq
         ccol_select_write: owner → owner_to_workers_cq  (same as chan_send_zc)
                          worker → workers_to_owner_cq

       Direction is resolved once, at the point this macro is evaluated.  If
       the calling thread's role can change, reconstruct the selectable before
       each ccol_select() call.

PARAMETERS
       ch   channel pointer.  Passing NULL produces a selectable that ccol_select()
            will reject with ccol_invalid_args.
       dir  ccol_select_read or ccol_select_write.

RETURN VALUE
       ccol_selectable value of type ccol_selectable_circq (a channel always resolves
       to one of its internal circular queues).

EXAMPLES
       /* Owner reads from two channels: */
       c_message_t msg;
       size_t idx;
       ccol_select_va(&msg, &idx,
           selectable_from_chan(ch0, ccol_select_read),
           selectable_from_chan(ch1, ccol_select_read));

       /* Owner waits for room to send on one channel: */
       c_message_t dummy = {0};
       ccol_select_va(&dummy, &idx,
           selectable_from_chan(ch, ccol_select_write));

NAME
       selectable_from_fd - build a ccol_selectable from a raw file descriptor
       (no read size limit)

SYNOPSIS
       #include <cthreadcomm.h>

       selectable_from_fd(fd, dir);
       selectable_from_fd_limited(fd, dir, max_bytes);

DESCRIPTION
       Constructs a ccol_selectable tagged-union value of type ccol_selectable_fd
       wrapping the given file descriptor with the specified I/O direction.

         ccol_select_read  — wake when the fd becomes readable (data available or
                            EOF / error condition)
         ccol_select_write — wake when the fd becomes writable (send buffer has room)

       Both macros expand to compound literals; they have no run-time overhead.

       selectable_from_fd() sets max_fd_read_bytes to 0 (unlimited).
       selectable_from_fd_limited() accepts an explicit max_bytes limit; if the
       data available on the fd exceeds that limit, ccol_select() returns
       ccol_msg_too_large without placing any data in buf (no partial reads).

       For non-blocking stream fds, the internal grow loop uses max_bytes + 1 as
       its ceiling so that a message of exactly max_bytes succeeds: the next read
       after filling the buffer returns EAGAIN, so total < capacity and the function
       returns ccol_success.  Only when total > max_bytes (more data than the limit)
       is ccol_msg_too_large returned.

       For ccol_select_read direction: when ccol_select() returns ccol_success,
       buf->data points to a heap buffer containing all data that was available on
       the fd at the moment of the read, and buf->size holds the byte count.  The
       caller gains ownership of buf->data and must call free() on it.  EOF yields
       buf->data = NULL, buf->size = 0.  This is the same ownership model as queue
       read wins.

       For ccol_select_write direction: buf is left untouched — ccol_select() only
       signals that the fd has room in its kernel send buffer.  The caller must
       subsequently call write(2) (symmetric with write-direction queue wins, where
       no queue slot is reserved).  max_bytes is ignored for write-direction fds.

       Read behaviour detail: ccol_select() inspects the fd with getsockopt(SO_TYPE)
       to determine whether it is a datagram socket (SOCK_DGRAM or SOCK_SEQPACKET).

       Datagram fds: a single read(2) is performed into a 66 KiB buffer.  This
       exceeds the maximum standard IPv4/IPv6 UDP payload (65,507 bytes), so the
       full datagram is always captured without truncation.  The grow-retry loop is
       suppressed for datagram fds regardless of O_NONBLOCK.  When max_bytes is set
       and the datagram size exceeds it, ccol_msg_too_large is returned.

       Stream fds and non-socket fds (pipes, timerfd, signalfd, etc.): a 4 KiB
       initial buffer is used.  If the fd has O_NONBLOCK, the read loop grows the
       buffer (up to max_bytes + 1 when a limit is set) until EAGAIN to drain all
       data that arrived before the wakeup.  If the fd is in blocking mode, exactly
       one read(2) call is made (safe because epoll already confirmed readiness).

       Mixing fd selectables with queue selectables in the same ccol_select() call
       is supported.  The implementation switches to an epoll(7)-based wait path
       when any fd selectable is present.  If only queue selectables are supplied
       the lighter condvar path is used instead.

PARAMETERS
       fd        An open file descriptor.  Must be >= 0.  Passing a negative value
                 produces a selectable that ccol_select() will reject with
                 ccol_invalid_args.
       dir       ccol_select_read or ccol_select_write.
       max_bytes (selectable_from_fd_limited only) Maximum bytes to read from the
                 fd on a ccol_select_read win.  0 is equivalent to selectable_from_fd
                 (unlimited).

RETURN VALUE
       ccol_selectable value (not a pointer).

EXAMPLES
       int pfd[2];
       pipe(pfd);  /* pfd[0] read end, pfd[1] write end */
       int src = 42;
       write(pfd[1], &src, sizeof(src));

       /* Wake when the pipe read end has data OR q has a message: */
       c_message_t msg;
       size_t idx;
       ccol_retval_t r = ccol_select_va(&msg, &idx,
           selectable_from_circq(q, ccol_select_read),
           selectable_from_fd(pfd[0], ccol_select_read));
       if (r == ccol_success && idx == 1) {
           /* fd won — data already read into msg.data */
           int val = *(int *)msg.data;
           free(msg.data);
       }

       /* Watch a socket for writability (buf untouched, no data read): */
       ccol_select_va(&msg, &idx,
           selectable_from_fd(sock_fd, ccol_select_write));

       /* Reject fd messages larger than 64 KiB: */
       r = ccol_select_va(&msg, &idx,
           selectable_from_fd_limited(pfd[0], ccol_select_read, 65536));
       if (r == ccol_msg_too_large) {
           /* data exceeded limit; fd kernel buffer still holds the bytes */
       }

       ccol_select, ccol_select_timed, ccol_select_va, ccol_select_timed_va,
       selectable_from_circq, selectable_from_dynq, selectable_from_chan

NAME
       ccol_select_va - wait for a message on an inline list of selectables
       (convenience macro)

SYNOPSIS
       #include <cthreadcomm.h>

       ccol_select_va(buf, ready_index, ...);
       ccol_select_timed_va(buf, ready_index, timeout_ms, ...);

DESCRIPTION
       ccol_select_va() builds a ccol_selectable array from the variadic arguments,
       computes the element count at compile time, and forwards everything to
       ccol_select().

       ccol_select_timed_va() does the same but calls ccol_select_timed() with the
       given timeout_ms (milliseconds; -1 = infinite, 0 = poll, positive = deadline).

       Each variadic argument must be a ccol_selectable value, typically produced
       by selectable_from_circq(), selectable_from_dynq(), selectable_from_chan(),
       or selectable_from_fd() with a ccol_select_read or ccol_select_write dir.
       All arguments are evaluated exactly once.

       The macros expand to GCC/Clang statement expressions and yield a value
       of type ccol_retval_t, so they can be used in assignments and conditions.

PARAMETERS
       buf          c_message_t * to receive the message.  On ccol_success the
                    caller owns buf->data.
       ready_index  size_t * set to the 0-based index of the ready selectable.
                    Valid only when ccol_success is returned.
       timeout_ms   (ccol_select_timed_va only) milliseconds to wait; -1 = infinite.
       ...          One or more ccol_selectable values.  At least one is required.

RETURN VALUE
       ccol_success            Message received.  *ready_index set; caller frees
                               buf->data.
       ccol_timed_out          (ccol_select_timed_va only) deadline elapsed without
                               any selectable becoming ready.
       ccol_invalid_args       Any argument is NULL/zero, or a selectable contains a
                               NULL queue pointer or unknown type tag.
       ccol_not_enough_memory  Waiter-node allocation failed (one node per selectable).
       ccol_msg_too_large      An fd selectable read exceeded its max_fd_read_bytes limit;
                               buf is left untouched; *ready_index IS set to the offending
                               fd's index; some or all fd bytes may already be consumed.
       ccol_unexpected_failure epoll/eventfd setup or the internal fd read(2) failed.

WARNINGS
       Uses a GCC/Clang statement expression (__extension__({ ... })).  Not
       valid under strict ISO C (-pedantic).  Compiles cleanly with -Wall
       -Wextra -Werror on both GCC and Clang without -pedantic.

NOTES
       The selectable array is a compound literal on the stack; no heap
       allocation occurs beyond that already performed by ccol_select_timed().

EXAMPLES
       c_message_t msg;
       size_t idx;

       /* Two circular queues and a channel, all read-wait: */
       ccol_retval_t r = ccol_select_va(&msg, &idx,
           selectable_from_circq(q0, ccol_select_read),
           selectable_from_circq(q1, ccol_select_read),
           selectable_from_chan(ch, ccol_select_read));
       if (r == ccol_success) {
           printf("ready selectable: %zu\n", idx);
           free(msg.data);
       }

       /* Mixed: write-wait on q0, read-wait on ch: */
       c_message_t dummy = {0};
       ccol_select_va(&dummy, &idx,
           selectable_from_circq(q0, ccol_select_write),
           selectable_from_chan(ch, ccol_select_read));

       /* Timed wait with 200 ms deadline: */
       r = ccol_select_timed_va(&msg, &idx, 200,
           selectable_from_circq(q0, ccol_select_read));
       if (r == ccol_timed_out) {
           puts("no message within 200 ms");
       }

       selectable_from_dynq, selectable_from_chan, selectable_from_fd,
       selectable_from_fd_limited, _selectable_from_chan

NAME
       clru_redeclare - restore type-tracking variables for a cache passed
       across scopes

SYNOPSIS
       #include <clrucache.h>

       clru_redeclare(name, KeyT, ValT);

DESCRIPTION
       The type-dispatching macros clru_get() and clru_set() rely on hidden
       companion variables created by clru_declare() or clru_construct(). When
       a clru_cache handle is passed to another function, those companion
       variables are absent. clru_redeclare() re-creates them so that all
       type-safe macros work correctly in the new scope.

PARAMETERS
       name  Existing clru_cache variable to re-attach type information to
       KeyT  Key type (must match the type used at construction)
       ValT  Value type (must match the type used at construction)

EXAMPLES
       void use_cache(clru_cache c) {
           clru_redeclare(c, int, double);
           double out = 0.0;
           if (clru_get(c, 42, &out) == ccol_success) {
               printf("%.2f\n", out);
           }
       }

NOTES
       Omitting clru_redeclare() before calling clru_get() or clru_set() in a
       new scope is the most common error when passing caches across function
       boundaries.

SEE ALSO
       clru_declare, clru_construct, clru_get, clru_set

------------------------------------------------------------------------------
NAME
       clru_get - retrieve a value by key (type-safe convenience wrapper)

SYNOPSIS
       #include <clrucache.h>

       ccol_retval_t clru_get(name, key, val_ptr);

DESCRIPTION
       Type-safe GNU C statement expression that serialises key into a
       cmap_pair and retrieves the stored value. The behaviour on success
       depends on the value type declared at cache construction:

       Non-char* val types: the value is copied directly into *val_ptr with
       no heap allocation. val_ptr is a pointer to ValT (e.g. double * for a
       cache<int, double>), NOT a cmap_pair *.

       char* val types: ownership of the heap-allocated string is transferred
       directly to the caller via *(char **)val_ptr. The caller must free()
       that pointer when done.

       Use clrucache_get_full() directly only if you need the raw cmap_pair
       output; clru_get() hides that entirely.

PARAMETERS
       name     Cache variable (must have companion type variables in scope,
                either from clru_construct / clru_declare or clru_redeclare).
                When passing a cache across function boundaries, call
                clru_redeclare() with a simple local alias first — the
                val-type detection relies on the name##__clru_val_type_var
                companion variable.
       key      Key expression. Must be an lvalue or a string literal; assign
                computed values to a variable first.
       val_ptr  Pointer to the value type (ValT *), NOT cmap_pair *.
                For non-char* val types: receives the value directly (e.g.
                double * for a double-valued cache); no heap allocation
                occurs.
                For char* val types: pointer to a char* variable; on success
                it is set to the heap-allocated string and the caller must
                free() it.

RETURN VALUE
       ccol_success           Value delivered via val_ptr.
       ccol_key_not_found     Key absent and no getter, or getter failed.
       ccol_invalid_args      cache or key_pair is NULL.
       ccol_not_enough_memory Internal allocation failed.

EXAMPLES
       /* Non-string value type: val_ptr is double *, not cmap_pair *;
          the value is copied directly into *val_ptr — no heap allocation */
       clru_construct(cache, int, double, 128, NULL, NULL, NULL);

       int k = 5;
       double v = 2.71;
       clru_set(cache, k, v);

       double out = 0.0;
       if (clru_get(cache, k, &out) == ccol_success) {
           printf("%.2f\n", out);
       }

       clru_destroy(cache);

       /* String value type: caller must free the returned string */
       clru_construct(str_cache, int, char *, 64, NULL, NULL, NULL);

       char *greeting = "hello";
       clru_set(str_cache, 1, greeting);

       char *result = NULL;
       if (clru_get(str_cache, 1, &result) == ccol_success) {
           printf("%s\n", result);
           free(result);
       }

       clru_destroy(str_cache);

WARNINGS
       For char* val types: the string placed in *(char **)val_ptr is a
       heap-allocated copy whose ownership is transferred to the caller.
       The caller MUST call free() on it when done; failing to do so is a
       memory leak.  Each call to clru_get for a char* value produces an
       independent allocation, so two successive calls for the same key
       yield two independent pointers that must each be freed separately.

       For non-char* val types: no heap allocation occurs.  The value is
       copied directly into *val_ptr; the caller must NOT call free() on it.

NOTES
       Uses a GNU C statement expression ({...}) — requires GCC or Clang.
       The cache never exposes a pointer into its internal storage; the value
       is always copied out before the mutex is released.

SEE ALSO
       clru_set, clrucache_get_full

------------------------------------------------------------------------------
NAME
       clru_set - store a value by key (type-safe convenience wrapper)

SYNOPSIS
       #include <clrucache.h>

       ccol_retval_t clru_set(name, key, val);

DESCRIPTION
       Type-safe GNU C statement expression that serialises key and val into
       cmap_pair values and calls clrucache_set_full().

       If a remote setter was configured at construction it is called
       synchronously before the cache is updated. On setter failure the
       function returns ccol_unexpected_failure and the cache is left unchanged.

PARAMETERS
       name  Cache variable
       key   Key expression. Must be an lvalue or a string literal.
       val   Value expression. Must be an lvalue or a string literal.

RETURN VALUE
       ccol_success             Value stored (remote setter succeeded, if any).
       ccol_unexpected_failure  Remote setter returned false; cache unchanged.
       ccol_invalid_args        Null pointer or zero-size argument.
       ccol_not_enough_memory   Memory allocation for the new value failed.

EXAMPLES
       clru_construct(cache, int, double, 128, NULL, NULL, NULL);

       int k = 42;
       double v = 3.14;
       clru_set(cache, k, v);

       clru_destroy(cache);

NOTES
       Uses a GNU C statement expression ({...}) — requires GCC or Clang.
       Both key and val must be lvalues; the macro takes their addresses
       internally. Assign literals or expressions to named variables first.

SEE ALSO
       clru_get, clrucache_set_full

Automatically included by cvector.h, chashmap.h, and cbstmap.h.  Any

NAME
       ccol_iter_declare - declare a typed iterator variable with RAII cleanup

SYNOPSIS
       #include <citerators.h>   /* or any container header */

       ccol_iter_declare(container, it);

DESCRIPTION
       Declares a cmap_iterator pointer named it bound to the type information
       of container.  The declaration attaches __attribute__((cleanup)) so the
       iterator is destroyed automatically when it leaves scope — no explicit
       ccol_iter_destroy() call is required.

       Also declares two hidden companion type variables that allow
       ccol_iter_key_ptr() and ccol_iter_val_ptr() to return correctly-typed
       pointers for container.

PARAMETERS
       container  Container variable previously declared with a *_declare /
                  *_construct macro.  Must be cvec, chmap, or cbmap.
       it         Name to use for the iterator variable.

NOTES
       container must be in scope and its type companion variables must be
       visible (same scope or a parent scope).
       Requires GCC or Clang (uses __attribute__((cleanup))).

EXAMPLES
       ccol_iter_declare(my_map, it);
       for (it = ccol_begin(my_map); it != NULL; it = ccol_iter_next(it)) {
           printf("%d\n", *ccol_iter_val_ptr(it));
       }

       ccol_begin, ccol_iter_next, ccol_for_each, ccol_iter_destroy

NAME

NAME
       ccol_for_each - iterate over all elements of any supported container

SYNOPSIS
       #include <citerators.h>   /* or any container header */

       ccol_for_each(container, it, { body });

DESCRIPTION
       Declares a cmap_iterator* variable named it with RAII cleanup, then
       loops from ccol_begin(container) to NULL using ccol_iter_next().  The
       body is executed once for each element; it gives typed access via
       ccol_iter_key_ptr(it) and ccol_iter_val_ptr(it).

       The iterator is destroyed automatically when the loop exits, whether
       by reaching the end, by break, or by return.

PARAMETERS
       container  A cvec, chmap, or cbmap variable.
       it         Name for the iterator variable.
       body       Compound statement executed per element (passed as variadic
                  argument; commas inside nested function calls are safe).

NOTES
       Do NOT modify the container inside the body — insertions and deletions
       invalidate the iterator and produce undefined behaviour.
       Requires GCC or Clang (uses statement expressions and cleanup).

EXAMPLES
       chmap_construct(freq, char*, int);
       /* ... populate ... */
       ccol_for_each(freq, it, {
           printf("%-12s %d\n", *ccol_iter_key_ptr(it),
                                *ccol_iter_val_ptr(it));
       });
       chmap_destroy(freq);

SEE ALSO
       ccol_iter_declare, ccol_begin, ccol_iter_next,
       ccol_iter_key_ptr, ccol_iter_val_ptr

NAME
       log_trace - emit a log line at TRACE level

SYNOPSIS
       #include <clogger.h>

       log_trace(clog lg, const char *fmt, ...);

DESCRIPTION
       Expands to a call to _clog_write() that captures __FILE__, __LINE__,
       and __func__ automatically and emits a log line at CLOG_TRACE level.
       No backtrace is appended.

       The output format is:
         ts=<ISO-8601-UTC> level=TRACE src=<file>:<line> func=<fn> [fields] msg=<text>

       The call is a no-op when lg is NULL or when the logger's current
       minimum level is above CLOG_TRACE.

PARAMETERS
       lg     Logger handle (clog). Passing NULL is a no-op.
       fmt    printf-style format string.
       ...    Format arguments (optional).

NOTES
       TRACE is the finest-grained level. Enable it only when diagnosing
       low-level behaviour; it generates high-volume output.
       All log_* macros share the same thread-safety guarantee: the entire
       format-and-write cycle is protected by the per-logger mutex, ensuring
       no partial lines are interleaved between threads.

EXAMPLES
       log_trace(lg, "entered function, x=%d", x);

       _clog_write, clog_open_fd_mp, clog_set_level

NAME
       log_error - emit a log line at ERROR level with a backtrace

SYNOPSIS
       #include <clogger.h>

       log_error(clog lg, const char *fmt, ...);

DESCRIPTION
       Expands to a call to _clog_write() at CLOG_ERROR level with automatic
       source location capture. After writing the structured log line, a
       stack backtrace is appended as tab-indented continuation lines:

         ts=... level=ERROR src=file.c:42 func=handle msg="db timeout"
         \t#0 ./myapp(handle+0x1a) [0x7f...]
         \t#1 ./myapp(main+0x42) [0x7f...]

       The continuation lines do not start with "ts=", allowing log
       aggregators to group them with the preceding ERROR record.

       Backtrace frames 0 and 1 (internal logger frames) are skipped so
       that frame #0 is the caller of log_error. Backtrace depth is capped
       at 64 frames.

       Backtrace requires the binary to be linked with -rdynamic for symbol
       resolution. On platforms without execinfo.h the backtrace lines are
       omitted silently and the log line is emitted normally.

PARAMETERS
       lg     Logger handle (clog). Passing NULL is a no-op.
       fmt    printf-style format string.
       ...    Format arguments (optional).

NOTES
       log_error does NOT terminate the process. Use log_fatal if process
       termination is desired (note: log_fatal also does not call abort() or
       exit(); that is the caller's responsibility).
       The backtrace write is also protected by the per-logger mutex, so
       the backtrace and the preceding log line always appear contiguously
       even under concurrent logging from multiple threads.

EXAMPLES
       log_error(lg, "failed to open %s: %s", path, strerror(errno));
       log_error(lg, "unexpected state %d in FSM", state);

SEE ALSO
       log_alert, log_fatal, log_warn, _clog_write, clog_open_fd_mp

NAME
       log_alert - emit a log line at ALERT level with a backtrace

SYNOPSIS
       #include <clogger.h>

       log_alert(clog lg, const char *fmt, ...);

DESCRIPTION
       Expands to a call to _clog_write() at CLOG_ALERT level with automatic
       source location capture. After writing the structured log line, a stack
       backtrace is appended in the same way as log_error():

         ts=... level=ALERT src=file.c:42 func=handle msg="disk full"
         \t#0 ./myapp(handle+0x1a) [0x7f...]
         \t#1 ./myapp(main+0x42) [0x7f...]

       ALERT maps to RFC 5424 syslog severity 1 ("alert — action must be taken
       immediately").  In CLOG_FMT_SYSLOG mode each backtrace frame is emitted
       as a separate syslog message using the same PRI and MSGID as the original
       record.

       ALERT sits between ERROR (severity 3) and FATAL (severity 0) in the
       clog_level_t ordering.  Use it for conditions that require immediate
       operator attention but do not necessarily terminate the process.

PARAMETERS
       lg     Logger handle (clog). Passing NULL is a no-op.
       fmt    printf-style format string.
       ...    Format arguments (optional).

NOTES
       Like log_error, log_alert does NOT terminate the process.
       Backtrace requires the binary to be linked with -rdynamic.  On platforms
       without execinfo.h the backtrace lines are silently omitted.
       The backtrace write is protected by the per-logger mutex, so the
       backtrace and the preceding log line always appear contiguously.

EXAMPLES
       log_alert(lg, "primary database unreachable, failing over");
       log_alert(lg, "disk usage at %d%%, writes may fail soon", pct);

SEE ALSO
       log_error, log_fatal, _clog_write, clog_open_fd_mp

NAME
       log_fatal - emit a log line at FATAL level with a backtrace

SYNOPSIS
       #include <clogger.h>

       log_fatal(clog lg, const char *fmt, ...);

DESCRIPTION
       Identical to log_error() except the severity level is CLOG_FATAL.
       A stack backtrace is appended after the log line.

       This macro does NOT terminate the process. It is the caller's
       responsibility to call abort(), exit(), or take other action after
       logging the fatal condition.

PARAMETERS
       lg     Logger handle (clog). Passing NULL is a no-op.
       fmt    printf-style format string.
       ...    Format arguments (optional).

NOTES
       Typical usage pattern:
         log_fatal(lg, "unrecoverable state: %s", reason);
         abort();

       The fact that log_fatal does not call abort() itself allows the caller
       to perform any final cleanup (flush queues, release locks, write a
       crash report) before terminating.

EXAMPLES
       log_fatal(lg, "assertion failed: invariant violated at step %d", step);
       abort();

       log_alert, log_error, _clog_write, clog_open_fd_mp
